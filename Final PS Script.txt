# Keep the machine awake for long-running exports. This avoids sleep/lock related disconnects.
function Set-ExecutionKeepAwake {
    Add-Type -TypeDefinition @"
using System;
using System.Runtime.InteropServices;
public static class NativePower {
    [DllImport("kernel32.dll", CharSet = CharSet.Auto, SetLastError = true)]
    public static extern uint SetThreadExecutionState(uint esFlags);
}
"@

    [uint32]$ES_CONTINUOUS        = 2147483648
    [uint32]$ES_SYSTEM_REQUIRED   = 1
    [uint32]$ES_AWAYMODE_REQUIRED = 64

    # Keep system active while script runs. Away mode is best-effort and may be ignored on some devices.
    [uint32]$executionStateFlags = $ES_CONTINUOUS -bor $ES_SYSTEM_REQUIRED -bor $ES_AWAYMODE_REQUIRED
    [void][NativePower]::SetThreadExecutionState($executionStateFlags)
    Write-Host "[INFO] Sleep prevention enabled for this run."
}

function Clear-ExecutionKeepAwake {
    if ("NativePower" -as [type]) {
        [uint32]$ES_CONTINUOUS = 2147483648
        [void][NativePower]::SetThreadExecutionState($ES_CONTINUOUS)
        Write-Host "[INFO] Sleep prevention cleared."
    }
}

# Enable no-sleep mode immediately; restore at script end.
Set-ExecutionKeepAwake
Register-EngineEvent -SourceIdentifier PowerShell.Exiting -Action { Clear-ExecutionKeepAwake } | Out-Null

# Ensure required modules are installed, and imports them. If import fails, error and exit early
function Import-ModuleQuietly {
    param(
        [Parameter(Mandatory)]
        [string]$ModuleName,
        [switch]$Strict
    )

    try {
        Import-Module $ModuleName -ErrorAction Stop -WarningAction SilentlyContinue | Out-Null
        return $true
    }
    catch {
        $message = [string]$_.Exception.Message
        if ($message -match '(?i)member .* is already present') {
            return $true
        }

        if ($Strict) {
            throw
        }

        return $false
    }
}

$requiredModules = @( 'ImportExcel', 'MicrosoftPowerBIMgmt' )
foreach ($module in $requiredModules) {
    $moduleImported = Import-ModuleQuietly -ModuleName $module
    if (-not $moduleImported) {
       Install-Module -Name $module -Scope CurrentUser -Force
       Import-ModuleQuietly -ModuleName $module -Strict # After install, require a successful import
    }
}


# ================================
# Read-Host With 60 Second Timeout
# ================================
function Read-HostWithTimeout {
    param(
        [string]$Prompt,
        [int]$TimeoutSeconds = 60
    )

    Write-Host -NoNewline "$Prompt "
    Write-Host "(waiting up to $TimeoutSeconds seconds):" -ForegroundColor DarkGray

    $inputBuffer = ""
    $endTime = (Get-Date).AddSeconds($TimeoutSeconds)

    while ((Get-Date) -lt $endTime) {
        if ([Console]::KeyAvailable) {
            $key = [Console]::ReadKey($true)
            # Enter pressed - stop reading
            if ($key.Key -eq "Enter") {
                Write-Host ""
                return $inputBuffer
            }
            # Backspace
            elseif ($key.Key -eq "Backspace") {
                if ($inputBuffer.Length -gt 0) {
                    $inputBuffer = $inputBuffer.Substring(0, $inputBuffer.Length - 1)
                    Write-Host -NoNewline "`b `b"
                }
            }
            else {
                $inputBuffer += $key.KeyChar
                Write-Host -NoNewline $key.KeyChar
            }
        }
        Start-Sleep -Milliseconds 100
    }

    Write-Host "Timeout reached - defaulting to Public" -ForegroundColor Yellow
    return ""
}

function Show-EnvironmentSelectionDialog {
    param(
        [int]$TimeoutSeconds = 60
    )

    Add-Type -AssemblyName System.Windows.Forms
    Add-Type -AssemblyName System.Drawing

    $form = New-Object System.Windows.Forms.Form
    $form.Text = "Select Power BI Environment"

    $form.FormBorderStyle = 'FixedDialog'
    $form.MaximizeBox = $false

    $label = New-Object System.Windows.Forms.Label
    $label.Text = "Select your Power BI cloud environment:"
    $label.AutoSize = $true
    $label.Location = New-Object System.Drawing.Point(20, 20)
    $form.Controls.Add($label)

    # ListBox for environment selection
    $listBox = New-Object System.Windows.Forms.ListBox
    $listBox.Location = New-Object System.Drawing.Point(20, 50)
    $listBox.Size = New-Object System.Drawing.Size(340, 150)

        'USGovHigh',
        'USGovMil'
    )
    
    foreach ($env in $environments) {
        [void]$listBox.Items.Add($env)
    }
    
    # Set default selection to Public
    $listBox.SelectedIndex = 0
    $form.Controls.Add($listBox)

    # Timeout label
    $timeoutLabel = New-Object System.Windows.Forms.Label
    $timeoutLabel.Text = "Timeout in $TimeoutSeconds seconds (defaults to Public)"
    $timeoutLabel.AutoSize = $true
    $timeoutLabel.Location = New-Object System.Drawing.Point(20, 210)
    $timeoutLabel.ForeColor = [System.Drawing.Color]::Gray
    $form.Controls.Add($timeoutLabel)

    # OK button
    $okButton = New-Object System.Windows.Forms.Button
    $okButton.Text = "OK"
    $okButton.Size = New-Object System.Drawing.Size(75, 30)
    $okButton.Location = New-Object System.Drawing.Point(205, 240)
    $okButton.Add_Click({
        $form.Tag = $listBox.SelectedItem
        $form.Close()
    })
    $form.Controls.Add($okButton)

    # Cancel button
    $cancelButton = New-Object System.Windows.Forms.Button
    $cancelButton.Text = "Cancel"
    $cancelButton.Size = New-Object System.Drawing.Size(75, 30)
    $cancelButton.Location = New-Object System.Drawing.Point(285, 240)
    $cancelButton.Add_Click({
        $form.Tag = 'Cancelled'
        $form.Close()
    })
    $form.Controls.Add($cancelButton)

    # Timer for timeout
    $timer = New-Object System.Windows.Forms.Timer
    $timer.Interval = $TimeoutSeconds * 1000
    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    })

    # Add FormClosing event handler
    $form.Add_FormClosing({
        param($formSender, $e)
        $timer.Stop()
        $timer.Dispose()
        
        # If Tag is not set (form closed without button click), mark as cancelled
        if (-not $form.Tag) {
            $form.Tag = 'Cancelled'
        }
    })

    # Handle double-click on list item (same as OK button)
    $listBox.Add_DoubleClick({
        $form.Tag = $listBox.SelectedItem
        $form.Close()
    })

    # Set default button and show
    $form.AcceptButton = $okButton
    $form.CancelButton = $cancelButton
    $timer.Start()
    [void]$form.ShowDialog()

    return $form.Tag
}

function Test-IsAuthError {
    param(
        [Parameter(Mandatory)]
        $ErrorRecord
    )

    $message = ($ErrorRecord | Out-String)
    $summary = Get-ErrorSummary -ErrorRecord $ErrorRecord
    $combined = "$message`n$summary"

    return ($combined -match '(?i)\b401\b|unauthorized|forbidden|token|expired|aadsts|authentication|invalid_grant|invalidtoken|access token is invalid')
}

function Get-ErrorSummary {
    param(
        [Parameter(Mandatory)]
        $ErrorRecord
    )

    $messages = @()

    if ($ErrorRecord.Exception) {
        $messages += $ErrorRecord.Exception.Message

        $innerException = $ErrorRecord.Exception.InnerException
        while ($innerException) {
            $messages += $innerException.Message
            $innerException = $innerException.InnerException
        }
    }

    if ($ErrorRecord.ErrorDetails -and $ErrorRecord.ErrorDetails.Message) {
        $messages += $ErrorRecord.ErrorDetails.Message
    }

    $messages = $messages | Where-Object { -not [string]::IsNullOrWhiteSpace($_) } | Select-Object -Unique
    if (-not $messages) {
        return ($ErrorRecord | Out-String).Trim()
    }

    return ($messages -join ' | ')
}

function Send-ReauthNotification {
    param(
        [string]$Context,
        [switch]$Force
    )

    if (-not $global:ReauthNotificationConfig.Enabled) {
        Write-Warning "Re-auth notification is disabled. Configure IMPACTIQ notification settings first."
        return $false
    }

    if ($global:ReauthNotificationConfig.Method -ne 'SMTP' -and $global:ReauthNotificationConfig.Method -ne 'OUTLOOK') {
        Write-Warning "Unknown IMPACTIQ_NOTIFY_METHOD '$($global:ReauthNotificationConfig.Method)'. Use SMTP or OUTLOOK."
        return
    }

    if (-not $Force -and $global:LastReauthNotificationSent) {
        $minutesSince = ((Get-Date) - $global:LastReauthNotificationSent).TotalMinutes
        if ($minutesSince -lt $global:ReauthNotificationConfig.RenotifyMinutes) {
            return
        }
    }

    $body = @"
ImpactIQ execution is paused because Power BI authentication expired.

Action required:
    1) Return to the ImpactIQ PowerShell window.
    2) Press Enter when you are ready to start browser sign-in.
    3) Complete sign-in in the browser window that opens.
    4) Script will resume automatically after successful login.

Host: $env:COMPUTERNAME
Time: $(Get-Date -Format o)
Context: $Context
"@

    try {
        if ($global:ReauthNotificationConfig.Method -eq 'OUTLOOK') {
            $outlook = $null
            $mailItem = $null

            try {
                $outlook = New-Object -ComObject Outlook.Application
                $mailItem = $outlook.CreateItem(0)
                $mailItem.Subject = $global:ReauthNotificationConfig.Subject
                $mailItem.Body = $body
                $mailItem.To = ($global:ReauthNotificationConfig.To -join ';')
                $mailItem.Send()
            }
            finally {
                if ($mailItem) { [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($mailItem) }
                if ($outlook) { [void][System.Runtime.InteropServices.Marshal]::ReleaseComObject($outlook) }
            }
        }
        else {
            $mailParams = @{
                SmtpServer = $global:ReauthNotificationConfig.SmtpServer
                Port       = $global:ReauthNotificationConfig.SmtpPort
                UseSsl     = $global:ReauthNotificationConfig.UseSsl
                From       = $global:ReauthNotificationConfig.From
                To         = ($global:ReauthNotificationConfig.To -join ',')
                Subject    = $global:ReauthNotificationConfig.Subject
                Body       = $body
                ErrorAction= 'Stop'
            }

            if ($global:ReauthNotificationConfig.Username -and $global:ReauthNotificationConfig.Password) {
                $securePassword = ConvertTo-SecureString $global:ReauthNotificationConfig.Password -AsPlainText -Force
                $credential = New-Object System.Management.Automation.PSCredential($global:ReauthNotificationConfig.Username, $securePassword)
                $mailParams.Credential = $credential
            }

            Send-MailMessage @mailParams
        }

        $global:LastReauthNotificationSent = Get-Date
        Write-Host "[INFO] Re-auth notification sent via $($global:ReauthNotificationConfig.Method) to: $($global:ReauthNotificationConfig.To -join ', ')"
        return $true
    }
    catch {
        Write-Warning "Failed to send re-auth notification via $($global:ReauthNotificationConfig.Method). $($_.Exception.Message)"
        return $false
    }
}

function Wait-ForPowerBIReauthentication {
    param(
        [string]$Context = 'Unknown operation'
    )

    Write-Warning "Authentication appears expired. Pausing until re-authentication succeeds. Context: $Context"
    Send-ReauthNotification -Context $Context -Force

    while ($true) {
        # Explicitly wait for user acknowledgement before opening a new browser sign-in prompt.
        Write-Host "[ACTION REQUIRED] ImpactIQ is paused for authentication." -ForegroundColor Yellow
        Write-Host "[ACTION REQUIRED] Press Enter in this window to begin Power BI re-authentication..." -ForegroundColor Yellow
        [void](Read-Host)

        try {
            Write-Host "[INFO] Launching Power BI sign-in now. Complete authentication in the browser..."
            Connect-PowerBI -Environment $LoginEnvironment

            if ($global:TokenRefreshStatusFilePath) {
                Set-Content -Path $global:TokenRefreshStatusFilePath -Value "$(Get-Date -Format o) - Re-authentication succeeded from paused state."
            }

            Write-Host "[INFO] Re-authentication succeeded. Resuming script."
            return
        }
        catch {
            Send-ReauthNotification -Context $Context
            Write-Warning "Re-authentication did not complete successfully. Resolve sign-in issues, then return and press Enter to try again."
        }
    }
}

function Invoke-PowerBIRestMethodWithRetry {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$Method,
        [Parameter(Mandatory)]
        [string]$Url,
        [string]$Body,
        [hashtable]$Headers
    )

    while ($true) {
        try {
            $invokeParams = @{
                Method      = $Method
                Url         = $Url
                ErrorAction = 'Continue'
                WarningAction = 'SilentlyContinue'
            }

            if ($PSBoundParameters.ContainsKey('Body')) {
                $invokeParams.Body = $Body
            }
            if ($PSBoundParameters.ContainsKey('Headers')) {
                $invokeParams.Headers = $Headers
            }

            $rawResult = @(MicrosoftPowerBIMgmt.Profile\Invoke-PowerBIRestMethod @invokeParams 2>&1)
            $errorRecords = @($rawResult | Where-Object { $_ -is [System.Management.Automation.ErrorRecord] })

            if ($errorRecords.Count -gt 0) {
                $firstError = $errorRecords[0]
                if (Test-IsAuthError -ErrorRecord $firstError) {
                    Wait-ForPowerBIReauthentication -Context "$Method $Url"
                    continue
                }

                Write-Warning "Power BI REST call failed unexpectedly and will be skipped. [$Method] $Url :: $(Get-ErrorSummary -ErrorRecord $firstError)"
                return $null
            }

            $resultValues = @($rawResult | Where-Object { $_ -isnot [System.Management.Automation.ErrorRecord] })
            if ($resultValues.Count -eq 0) {
                return $null
            }

            if ($resultValues.Count -eq 1) {
                return [string]$resultValues[0]
            }

            return (($resultValues | ForEach-Object { [string]$_ }) -join [Environment]::NewLine)
        }
        catch {
            if (Test-IsAuthError -ErrorRecord $_) {
                Wait-ForPowerBIReauthentication -Context "$Method $Url"
                continue
            }

            Write-Warning "Power BI REST call failed unexpectedly and will be skipped. [$Method] $Url :: $(Get-ErrorSummary -ErrorRecord $_)"
            return $null
        }
    }
}

# Route all existing Invoke-PowerBIRestMethod calls through auth-aware retry logic.
Set-Alias -Name Invoke-PowerBIRestMethod -Value Invoke-PowerBIRestMethodWithRetry -Scope Script -Force

function Invoke-AuthenticatedRestMethodWithRetry {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)]
        [string]$Uri,
        [Parameter(Mandatory)]
        [string]$Method,
        [hashtable]$Headers,
        [string]$Body,
        [string]$OutFile
    )

    $script:LastAuthenticatedRestErrorSummary = $null

    $attemptNumber = 0
    $authRetryCount = 0
    $maxAuthRetryCount = 1

    while ($true) {
        $attemptNumber++

        try {
            Update-AuthorizationHeaderFromCurrentToken -Headers $Headers

            $invokeParams = @{
                Uri         = $Uri
                Method      = $Method
                ErrorAction = 'Stop'
            }

            if ($PSBoundParameters.ContainsKey('Headers')) {
                $invokeParams.Headers = $Headers
            }
            if ($PSBoundParameters.ContainsKey('Body')) {
                $invokeParams.Body = $Body
            }
            if ($PSBoundParameters.ContainsKey('OutFile')) {
                $invokeParams.OutFile = $OutFile
            }

            $result = Invoke-RestMethod @invokeParams
            $script:LastAuthenticatedRestErrorSummary = $null
            return $result
        }
        catch {
            if (Test-IsAuthError -ErrorRecord $_) {
                $authRetryCount++
                $authErrorSummary = Get-ErrorSummary -ErrorRecord $_
                $script:LastAuthenticatedRestErrorSummary = $authErrorSummary

                if ($authRetryCount -gt $maxAuthRetryCount) {
                    Write-Output "[WARN] Auth-like error persisted after re-authentication for [$Method] $Uri. Treating as non-recoverable and skipping call. Details: $authErrorSummary"
                    return $null
                }

                Write-Output "[WARN] Authentication error during authenticated REST call. Attempt $attemptNumber will pause for re-authentication. [$Method] $Uri :: $authErrorSummary"
                Wait-ForPowerBIReauthentication -Context "$Method $Uri"

                Update-AuthorizationHeaderFromCurrentToken -Headers $Headers
                continue
            }

            $httpStatusCode = $null
            try {
                if ($_.Exception.Response -and $_.Exception.Response.StatusCode) {
                    $httpStatusCode = [int]$_.Exception.Response.StatusCode
                }
            }
            catch {
            }

            $baseErrorSummary = Get-ErrorSummary -ErrorRecord $_
            $errorSummary = if ($httpStatusCode) { "HTTP $httpStatusCode - $baseErrorSummary" } else { $baseErrorSummary }
            $script:LastAuthenticatedRestErrorSummary = $errorSummary

            Write-Warning "Authenticated REST call failed and will be skipped. [$Method] $Uri :: $errorSummary"
            return $null
        }
    }
}

# Function to refresh the token in a background job
# Start a background job that refreshes the token. Pass the environment so child job can reconnect correctly.
function Start-TokenRefreshJob {
    param(
        [string]$Environment,
        [string]$BaseFolderPath,
        [string]$TokenRefreshStatusFilePath
    )

    $jobScript = {
        param($EnvArg, $BaseFolderPathArg, $StatusFilePathArg)

        function Import-PowerBIModuleQuietly {
            param([string]$ModuleName)

            if (Get-Module -Name $ModuleName) {
                return
            }

            try {
                Import-Module $ModuleName -ErrorAction Stop -WarningAction SilentlyContinue | Out-Null
            }
            catch {
                if ($_.Exception.Message -match 'member .* is already present') {
                    return
                }
                if ($StatusFilePathArg) {
                    Set-Content -Path $StatusFilePathArg -Value "$(Get-Date -Format o) - Module import warning for $ModuleName. $($_.Exception.Message)"
                }
            }
        }

        Import-PowerBIModuleQuietly -ModuleName 'MicrosoftPowerBIMgmt.Profile'
        Import-PowerBIModuleQuietly -ModuleName 'MicrosoftPowerBIMgmt'

        $tokenDirectory = Join-Path -Path $BaseFolderPathArg -ChildPath 'Config'
        if (-not (Test-Path -Path $tokenDirectory)) {
            New-Item -Path $tokenDirectory -ItemType Directory -Force | Out-Null
        }
        $tokenFilePath = Join-Path -Path $tokenDirectory -ChildPath 'PowerBI_TempAccessToken.txt'

        function Update-AccessToken {
            param(
                [string]$Environment,
                [string]$TokenFilePath,
                [string]$StatusFilePath
            )

            try {
                # Try silent token refresh first from current session context.
                $tokenObj = Get-PowerBIAccessToken -ErrorAction Stop
                $token = $tokenObj.Authorization -replace 'Bearer ', ''
                Set-Content -Path $TokenFilePath -Value $token
                Set-Content -Path $StatusFilePath -Value "$(Get-Date -Format o) - Token refresh succeeded (silent)."
                return
            }
            catch {
                try {
                    # Fallback: reconnect if silent refresh is unavailable.
                    if ($Environment) {
                        Connect-PowerBIServiceAccount -Environment $Environment | Out-Null
                    } else {
                        Connect-PowerBIServiceAccount | Out-Null
                    }

                    $tokenObj = Get-PowerBIAccessToken -ErrorAction Stop
                    $token = $tokenObj.Authorization -replace 'Bearer ', ''
                    Set-Content -Path $TokenFilePath -Value $token
                    Set-Content -Path $StatusFilePath -Value "$(Get-Date -Format o) - Token refresh succeeded (reconnect)."
                }
                catch {
                    # Persist failure details so operator can inspect without watching terminal.
                    Set-Content -Path $StatusFilePath -Value "$(Get-Date -Format o) - Token refresh FAILED. Re-authentication may be required. $($_.Exception.Message)"
                }
            }
        }

        while ($true) {
            Start-Sleep -Seconds 3300  # Sleep for 55 minutes
            Update-AccessToken -Environment $EnvArg -TokenFilePath $tokenFilePath -StatusFilePath $StatusFilePathArg
        }
    }

    # Start the job and pass the environment as an argument
    Start-Job -ScriptBlock $jobScript -Name "TokenRefreshJob" -ArgumentList $Environment, $BaseFolderPath, $TokenRefreshStatusFilePath | Out-Null
}

# Start the background job to refresh the token
Start-TokenRefreshJob -Environment $LoginEnvironment -BaseFolderPath $global:BaseFolderPath -TokenRefreshStatusFilePath $global:TokenRefreshStatusFilePath

# Function to get the current access token
function Get-CurrentAccessToken {
    if (-not (Test-Path -Path $global:TokenFilePath)) {
        Wait-ForPowerBIReauthentication -Context 'Missing token file'
    }

    $global:accessToken = Get-Content -Path $global:TokenFilePath -ErrorAction SilentlyContinue
    if ([string]::IsNullOrWhiteSpace($global:accessToken)) {
        Wait-ForPowerBIReauthentication -Context 'Empty token value'
        $global:accessToken = Get-Content -Path $global:TokenFilePath -ErrorAction Stop
    }

    return $global:accessToken
}

    Write-Host "=================================================================" -ForegroundColor Cyan
    Write-Host "  A checkpoint from today's run was found." -ForegroundColor Cyan
    Write-Host "=================================================================" -ForegroundColor Cyan
    $phaseNames = @(
        'Phase1EnvDetailComplete',
        'Phase2ModelBackupComplete',
        'Phase3ReportBackupComplete',
        'Phase4ReportDetailComplete',
        'Phase5ModelDetailComplete',
        'Phase6DataflowComplete'
    )
    foreach ($p in $phaseNames) {
        $done = [bool]$savedCheckpoint.$p
        Write-Host ("  {0,-35}: {1}" -f $p, $(if ($done) { 'DONE' } else { 'incomplete' })) `
            -ForegroundColor $(if ($done) { 'Green' } else { 'Yellow' })
    }
    Write-Host ""
    $resumeChoice = $null
    $resolvedResumeMode = if ($script:ResumeModeOverride) { $script:ResumeModeOverride } else { $script:ImpactIQSettings.ResumeMode }

    if ($script:NoPromptsMode) {
        if ($resolvedResumeMode -in @('resume', 'fresh')) {
            $resumeChoice = if ($resolvedResumeMode -eq 'resume') { 'Y' } else { 'N' }
            Write-Output "[CONFIG] Resume mode: $resolvedResumeMode"
        }
        else {
            $resumeChoice = 'Y'
            Write-Warning "No-prompts mode enabled but ResumeMode is invalid/empty. Defaulting to resume."
        }
    }
    else {
        while ($true) {
            $defaultChoice = if ($resolvedResumeMode -eq 'fresh') { 'N' } else { 'Y' }
            $resumeChoiceInput = Read-Host "[ACTION REQUIRED] Resume from checkpoint? (Y/N, default $defaultChoice)"

            if ([string]::IsNullOrWhiteSpace($resumeChoiceInput)) {
                $resumeChoice = $defaultChoice
                break
            }

            if ($resumeChoiceInput -match '^[YyNn]$') {
                $resumeChoice = $resumeChoiceInput
                break
            }

            Write-Host "Please enter Y or N." -ForegroundColor Yellow
        }
    }

    if ($resumeChoice -match '^[Yy]') {
        $resumeMode = $true
        foreach ($p in $phaseNames) { $checkpoint[$p] = [bool]$savedCheckpoint.$p }
        Write-Output "[RESUME] Resuming from checkpoint. Completed phases will be skipped."
        $script:ImpactIQSettings.ResumeMode = 'resume'
    } else {
        Write-Output "[RESUME] Starting fresh run. Checkpoint will be overwritten."
        $script:ImpactIQSettings.ResumeMode = 'fresh'
    }

    Save-ImpactIQSettings -Settings $script:ImpactIQSettings
}

function Get-LatestImpactIQCacheFile {
    param(
        [Parameter(Mandatory)]
        [string]$FileName,
        [string]$ExcludeFolder
    )

    $cacheRoot = Join-Path $configFolder 'Cache'
    if (-not (Test-Path $cacheRoot)) {
        return $null
    }

    $cacheDirectories = Get-ChildItem -Path $cacheRoot -Directory -ErrorAction SilentlyContinue |
        Where-Object { $_.Name -match '^\d{4}-\d{2}-\d{2}$' }

    if (-not [string]::IsNullOrWhiteSpace($ExcludeFolder)) {
        $cacheDirectories = @($cacheDirectories | Where-Object { $_.FullName -ne $ExcludeFolder })
    }

    foreach ($cacheDirectory in ($cacheDirectories | Sort-Object Name -Descending)) {
        $candidatePath = Join-Path $cacheDirectory.FullName $FileName
        if (Test-Path $candidatePath) {
            return $candidatePath
        }
    }

    return $null
}

function Resolve-TargetReportScopeFromCache {
    param(
        [string[]]$IncludeReportIds,
        [string[]]$IncludeReportNames,
        [string]$CurrentCacheFolder
    )

    $hasIdFilters = $IncludeReportIds -and $IncludeReportIds.Count -gt 0
    $hasNameFilters = $IncludeReportNames -and $IncludeReportNames.Count -gt 0

    if (-not $hasIdFilters -and -not $hasNameFilters) {
        return [PSCustomObject]@{
            FilterActive = $false
            ReportIds = @()
            DatasetIds = @()
            WorkspaceIds = @()
            ReportRows = @()
        }
    }

    $reportsCachePath = Get-LatestImpactIQCacheFile -FileName 'reportsInfo.csv' -ExcludeFolder $CurrentCacheFolder
    if (-not $reportsCachePath) {
        return [PSCustomObject]@{
            FilterActive = $true
            ReportIds = @($IncludeReportIds)
            DatasetIds = @()
            WorkspaceIds = @()
            ReportRows = @()
        }
    }

    $cachedReports = @(Import-Csv -Path $reportsCachePath)
    $matchedReports = @()

    if ($hasIdFilters) {
        $matchedReports += @($cachedReports | Where-Object { $IncludeReportIds -contains [string]$_.ReportId })
    }

    if ($hasNameFilters) {
        foreach ($reportNameFilter in $IncludeReportNames) {
            $usesWildcard = $reportNameFilter -match '[*?]'
            if ($usesWildcard) {
                $matchedReports += @($cachedReports | Where-Object { [string]$_.ReportName -like $reportNameFilter })
            }
            else {
                $matchedReports += @($cachedReports | Where-Object { [string]$_.ReportName -ieq $reportNameFilter })
            }
        }
    }

    $matchedReports = @($matchedReports | Group-Object ReportId | ForEach-Object { $_.Group[0] })

    $targetReportIds = @($matchedReports | ForEach-Object { [string]$_.ReportId } | Where-Object { -not [string]::IsNullOrWhiteSpace($_) })
    if ($hasIdFilters) {
        $targetReportIds = @($targetReportIds + $IncludeReportIds | Where-Object { -not [string]::IsNullOrWhiteSpace($_) } | Select-Object -Unique)
    }

    $targetDatasetIds = @($matchedReports | ForEach-Object { [string]$_.DatasetId } | Where-Object { -not [string]::IsNullOrWhiteSpace($_) } | Select-Object -Unique)
    $targetWorkspaceIds = @(
        @($matchedReports | ForEach-Object { [string]$_.WorkspaceId }) +
        @($matchedReports | ForEach-Object { [string]$_.DatasetWorkspaceId })
    ) | Where-Object { -not [string]::IsNullOrWhiteSpace($_) } | Select-Object -Unique

    return [PSCustomObject]@{
        FilterActive = $true
        ReportIds = $targetReportIds
        DatasetIds = $targetDatasetIds
        WorkspaceIds = $targetWorkspaceIds
        ReportRows = $matchedReports
    }
}

if ($script:TargetReportScope.FilterActive -and $script:TargetReportScope.ReportRows.Count -gt 0) {

}

function Get-StringSha256 {
    param(
        [Parameter(Mandatory)]
        [string]$Text
    )

    $sha = [System.Security.Cryptography.SHA256]::Create()
    try {
        $bytes = [System.Text.Encoding]::UTF8.GetBytes($Text)
        $hashBytes = $sha.ComputeHash($bytes)
        return ([System.BitConverter]::ToString($hashBytes)).Replace('-', '').ToLowerInvariant()
    }
    finally {
        $sha.Dispose()
    }
}

function Load-ReportExportState {
    if (-not (Test-Path $script:ReportExportStatePath)) {
        return @{}
    }

    try {
        $rawState = Get-Content -Path $script:ReportExportStatePath -Raw -ErrorAction Stop | ConvertFrom-Json
        $stateMap = @{}

        if ($rawState -and $rawState.Entries) {
            foreach ($entry in $rawState.Entries) {
                if ($entry -and -not [string]::IsNullOrWhiteSpace([string]$entry.Key)) {
                    $stateMap[[string]$entry.Key] = $entry
                }
            }
        }

        return $stateMap
    }
    catch {
        Write-Warning "Failed to read report export state from $($script:ReportExportStatePath). Starting with empty state. $($_.Exception.Message)"
        return @{}
    }
}

        } | ConvertTo-Json -Depth 6 | Set-Content -Path $script:ReportExportStatePath -Encoding UTF8 -Force

        $script:ReportExportStateDirty = $false
    }
    catch {

    }
}

function Test-IsKnownNoModelReport {
    param(
        [Parameter(Mandatory)]
        [string]$ReportId,
        [Parameter(Mandatory)]
        [string]$ReportName
    )

    if ($script:KnownNoModelReportIds -contains $ReportId) {
        return $true
    }

    foreach ($nameFilter in $script:KnownNoModelReportNames) {
        if ([string]::IsNullOrWhiteSpace($nameFilter)) {
            continue

        if ($usesWildcard) {
            if ($ReportName -like $nameFilter) {

            }

        elseif ($ReportName -ieq $nameFilter) {
            return $true
        }
    }

    return $false

function New-ControlledRunspacePool {
    <#
    .SYNOPSIS
    Creates a runspace pool for parallel workspace metadata collection.
    
    .PARAMETER MinThreads
    Minimum thread count (default 2).
    
    .PARAMETER MaxThreads
    Maximum thread count (default 4 for metadata collection, can be tuned based on throttle limits).
    
    .PARAMETER InitialState
    Optional InitialSessionState to inject functions/variables into each worker runspace.
    #>
    param(
        [int]$MinThreads = 2,
        [int]$MaxThreads = 4,
        [System.Management.Automation.Runspaces.InitialSessionState]$InitialState = $null
    )

    if ($InitialState) {
        $runspacePool = [System.Management.Automation.Runspaces.RunspaceFactory]::CreateRunspacePool(
            $MinThreads, 
            $MaxThreads,
            $InitialState,
            $Host
        )
    } else {
        $runspacePool = [System.Management.Automation.Runspaces.RunspaceFactory]::CreateRunspacePool(
            $MinThreads, 
            $MaxThreads
        )
    }
    
    $runspacePool.Open()
    
    return $runspacePool
}

function Invoke-ParallelWorkspaceMetadata {
    <#
    .SYNOPSIS
    Parallel executor for workspace metadata collection with per-worker isolation.
    
    .PARAMETER Workspaces
    Array of workspace objects to process.
    
    .PARAMETER RunspacePool
    Runspace pool instance.
    
    .PARAMETER WorkerScriptBlock
    Script block to execute per workspace (receives $workspace as $_ argument).
    
    .NOTES
    Each worker collects results in isolated collections; caller merges results after execution.
    #>
    param(
        [Parameter(Mandatory)]
        [psobject[]]$Workspaces,
        
        [Parameter(Mandatory)]
        [System.Management.Automation.Runspaces.RunspacePool]$RunspacePool,
        
        [Parameter(Mandatory)]
        [scriptblock]$WorkerScriptBlock
    )

    $jobs = @()
    
    foreach ($workspace in $Workspaces) {
        $ps = [System.Management.Automation.PowerShell]::Create()
        $ps.RunspacePool = $RunspacePool
        
        # Add shared context before invoking worker script
        $ps.AddScript({
            # Import global endpoints and access token provider
            param($WorkerWorkspace)
        }).AddArgument($workspace) | Out-Null
        
        # Append worker logic
        $ps.AddScript($WorkerScriptBlock).AddArgument($workspace) | Out-Null
        
        $asyncResult = $ps.BeginInvoke()
        $jobs += @{
            PowerShell = $ps
            AsyncResult = $asyncResult
            Workspace = $workspace
            WorkerIndex = $jobs.Count
        }
    }

    # Collect results with timeout per job
    $results = @()
    foreach ($job in $jobs) {
        try {
            # Wait with 60-second timeout per workspace job
            if ($job.AsyncResult.AsyncWaitHandle.WaitOne(60000)) {
                $workerOutput = $job.PowerShell.EndInvoke($job.AsyncResult)
                $results += $workerOutput
            }
            else {
                Write-Warning "Workspace [$($job.Workspace.WorkspaceName)] metadata collection timed out (60s)."
            }
        }
        catch {
            Write-Warning "Worker error on workspace [$($job.Workspace.WorkspaceName)]: $(Get-ErrorSummary -ErrorRecord $_)"
        }
        finally {
            $job.PowerShell.Dispose()
        }
    }

    return $results
}

function New-WorkspaceMetadataWorker {
    <#
    .SYNOPSIS
    Factory for per-workspace metadata collection scriptblock.
    Used to parallelize report/dataset/dataflow fetch.
    
    .NOTES
    Worker returns PSCustomObject with:
      - WorkspaceInfo, DatasetsInfo, ReportsInfo, DataflowsInfo collections per worker
      - Errors captured for logging after parallel execution completes
    #>
    
    # This scriptblock is executed inside each runspace
    $workerScript = {
        param(
            [psobject]$Workspace,
            [hashtable]$GlobalContext
        )
        
        $workerResults = @{
            WorkspaceName = $Workspace.WorkspaceName
            WorkspaceId = $Workspace.WorkspaceId
            DatasetsInfo = @()
            ReportsInfo = @()
            DatasetSourcesInfo = @()
            DataflowsInfo = @()
            DataflowSourcesInfo = @()
            Errors = @()
        }
        
        try {
            # Worker will call API and collect results
            # Caller will invoke this with shared context
            $workerResults
        }
        catch {
            $workerResults.Errors += "Worker initialization failed: $_"
            $workerResults
        }
    }
    
    return $workerScript
}

function Convert-ToExcelSafeText {
    param(
        [Parameter()]
        [object]$Value
    )

    if ($null -eq $Value) {
        return $null
    }

    $text = [string]$Value
    $text = [regex]::Replace($text, '[^\u0009\u000A\u000D\u0020-\uD7FF\uE000-\uFFFD]', '')

    if ($text.Length -gt 32767) {
        $text = $text.Substring(0, 32767)
    }

    return $text
}

function Convert-ObjectsToExcelSafe {
    param(
        [Parameter()]
        [object[]]$Rows
    )

    if (-not $Rows -or $Rows.Count -eq 0) {
        return @()
    }

    $safeRows = @()
    foreach ($row in $Rows) {
        if ($null -eq $row) {
            continue
        }

        $safeRow = [ordered]@{}
        foreach ($property in $row.PSObject.Properties) {
            $propertyName = $property.Name
            $propertyValue = $property.Value

            if ($null -eq $propertyValue) {
                $safeRow[$propertyName] = $null
            }
            elseif ($propertyValue -is [string]) {
                $safeRow[$propertyName] = Convert-ToExcelSafeText -Value $propertyValue
            }
            elseif ($propertyValue -is [ValueType]) {
                $safeRow[$propertyName] = $propertyValue
            }
            else {
                $safeRow[$propertyName] = Convert-ToExcelSafeText -Value $propertyValue
            }
        }

        $safeRows += [PSCustomObject]$safeRow
    }

    return $safeRows
}

# Function to rename properties in objects and handle duplicates
function Rename-Properties {
    param ($object, $renameMap)
    $newObject = New-Object PSObject
    foreach ($originalName in $renameMap.Keys) {
        $newPropertyName = $renameMap[$originalName]
        $propertyValue = if ($object.PSObject.Properties[$originalName]) { $object.$originalName } else { $null }
        if ($newObject.PSObject.Properties[$newPropertyName]) { $newPropertyName += "_duplicate" }
        $newObject | Add-Member -MemberType NoteProperty -Name $newPropertyName -Value $propertyValue
    }
    foreach ($property in $object.PSObject.Properties) {
        if (-not $renameMap.ContainsKey($property.Name)) {
            $newObject | Add-Member -MemberType NoteProperty -Name $property.Name -Value $property.Value
        }
    }
    return $newObject
}

    $lbl.AutoSize       = $true

    $form.Controls.Add($lbl)

    # Search label
    $lblSearch          = New-Object System.Windows.Forms.Label
    $lblSearch.Text     = "Search:"
    $lblSearch.AutoSize = $true
    $lblSearch.Location = New-Object System.Drawing.Point(12,40)
    $form.Controls.Add($lblSearch)

    # Search TextBox
    $txtSearch          = New-Object System.Windows.Forms.TextBox
    $txtSearch.Location = New-Object System.Drawing.Point(70,37)
    $txtSearch.Size     = New-Object System.Drawing.Size(422,23)
    $form.Controls.Add($txtSearch)

    # CheckedListBox
    $clb                = New-Object System.Windows.Forms.CheckedListBox
    $clb.Location       = New-Object System.Drawing.Point(12,70)
	$clb.Size           = New-Object System.Drawing.Size(480,380)
    $clb.CheckOnClick   = $true
    $clb.Sorted         = $true

    # Store all workspace items for filtering
    $allWorkspaceItems = @()
    foreach ($ws in $Workspaces) {
        $display = "{0} ({1})" -f $ws.name, $ws.id
        $allWorkspaceItems += [pscustomobject]@{ Display=$display; Id=$ws.id; Name=$ws.name }
    }

    # Maintain persistent checked state across searches
    $persistentCheckedIds = @{}

    # Initial population
    foreach ($item in $allWorkspaceItems) {
        [void]$clb.Items.Add($item)
    }
    $clb.DisplayMember = 'Display'
    $form.Controls.Add($clb)

	# Track when items are checked/unchecked
    $clb.Add_ItemCheck({
        param($checkListSender, $e)
        $item = $clb.Items[$e.Index]
        if ($e.NewValue -eq 'Checked') {
            $persistentCheckedIds[$item.Id] = $true
        } elseif ($e.NewValue -eq 'Unchecked') {
            $persistentCheckedIds.Remove($item.Id)
        }
    })

	# Search filter logic
    $txtSearch.Add_TextChanged({
        $searchText = $txtSearch.Text
        $clb.BeginUpdate()
        
        $clb.Items.Clear()
        
        # Escape search text once for reuse in loop
        $escapedSearchText = [regex]::Escape($searchText)
        
        # Filter and re-add items
        foreach ($item in $allWorkspaceItems) {
            if ([string]::IsNullOrWhiteSpace($searchText) -or 
                $item.Display -imatch $escapedSearchText) {
                $index = $clb.Items.Add($item)
                # Restore checked state from persistent store
                if ($persistentCheckedIds.ContainsKey($item.Id)) {
                    $clb.SetItemChecked($index, $true)
                }
            }
        }
        
        $clb.EndUpdate()
    })

    # "Select All" checkbox
    $chkAll            = New-Object System.Windows.Forms.CheckBox
    $chkAll.Text       = "Select All"
    $chkAll.AutoSize   = $true
    $chkAll.Location   = New-Object System.Drawing.Point(12,460)
    $chkAll.Add_CheckedChanged({
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $chkAll.Checked)
        }
    })
    $form.Controls.Add($chkAll)

    # "Clear All" button
    $btnClearAll       = New-Object System.Windows.Forms.Button
    $btnClearAll.Text  = "Clear All"
    $btnClearAll.Width = 80
    $btnClearAll.Location = New-Object System.Drawing.Point(90,458)
    $btnClearAll.Add_Click({
        # Uncheck all items - ItemCheck events will update persistent store
        for ($i=0; $i -lt $clb.Items.Count; $i++) {
            $clb.SetItemChecked($i, $false)
        }
        $chkAll.Checked = $false
    })
    $form.Controls.Add($btnClearAll)

    # "Include My Workspace" checkbox
    $chkMy             = New-Object System.Windows.Forms.CheckBox
    $chkMy.Text        = "Include 'My Workspace'"
    $chkMy.AutoSize    = $true
    $chkMy.Location    = New-Object System.Drawing.Point(184,460)
    $chkMy.Checked     = $false
    $form.Controls.Add($chkMy)

    # OK button
    $okBtn             = New-Object System.Windows.Forms.Button
    $okBtn.Text        = "OK"
    $okBtn.Width       = 100
    $okBtn.Location    = New-Object System.Drawing.Point(286,500)
    $okBtn.Add_Click({
        $form.Tag = 'OK'
        $form.Close()
    })
    $form.Controls.Add($okBtn)

    # Cancel button
    $cancelBtn         = New-Object System.Windows.Forms.Button
    $cancelBtn.Text    = "Cancel"
    $cancelBtn.Width   = 100
    $cancelBtn.Location= New-Object System.Drawing.Point(392,500)
    $cancelBtn.Add_Click({
        $form.Tag = 'Cancel'
        $form.Close()
    })
    $form.Controls.Add($cancelBtn)

    # Timer (auto-close after 10 minutes)
    $timer = New-Object System.Windows.Forms.Timer
    $timer.Interval = 600000
    $timer.Add_Tick({
        $timer.Stop()
        $form.Tag = 'Timeout'
        $form.Close()
    }.GetNewClosure())
    $timer.Start()

    [void]$form.ShowDialog()

    # Gather selections from persistent store (to include items not currently visible due to search)
    $selectedIds = @($persistentCheckedIds.Keys)

    if ($form.Tag -eq 'Cancel') {
        throw "User cancelled workspace selection."
    }

    $timedOut = ($form.Tag -eq 'Timeout')

    # Include My Workspace only if checked… except on timeout, then force include
    $includeMy = if ($timedOut) { $true } else { $chkMy.Checked }

    # If nothing selected from workspace list:
    # - If "Include My Workspace" is checked, process only My Workspace (empty array for regular workspaces)
    # - Otherwise, default to all workspaces (backwards compatibility with timeout/cancel behavior)
    if ($selectedIds.Count -eq 0 -and -not $includeMy) {
        $selectedIds = $Workspaces.id
    }

    return [pscustomobject]@{
        SelectedWorkspaceIds = $selectedIds
        IncludeMyWorkspace   = $includeMy
        TimedOut             = $timedOut
    }
}

# ================================================================
# Controlled Parallelism: Workspace Metadata Collection 
# ================================================================
# Phase 1 parallelism uses runspace pool with initialized functions.
# Workers use simplified Invoke-RestMethod directly to avoid module dependencies.

Write-Output "[CONFIG] Controlled Parallelism setting: Enabled=$($script:EnableControlledParallelism), MaxThreads=$($script:ControlledParallelismMaxThreads)"

if ($script:EnableControlledParallelism -and $workspacesInfo.Count -gt 1) {
    Write-Output "[PERF] Initializing runspace pool for workspace metadata collection ($($workspacesInfo.Count) workspaces, $($script:ControlledParallelismMaxThreads) threads)..."
    
    # Inject Rename-Properties into each worker runspace via InitialSessionState
    $iss = [System.Management.Automation.Runspaces.InitialSessionState]::CreateDefault()
    $renamePropsFunctionBody = @'
    param ($object, $renameMap)
    $newObject = New-Object PSObject
    foreach ($originalName in $renameMap.Keys) {
        $newPropertyName = $renameMap[$originalName]
        $propertyValue = if ($object.PSObject.Properties[$originalName]) { $object.$originalName } else { $null }
        if ($newObject.PSObject.Properties[$newPropertyName]) { $newPropertyName += "_duplicate" }
        $newObject | Add-Member -MemberType NoteProperty -Name $newPropertyName -Value $propertyValue
    }
    foreach ($property in $object.PSObject.Properties) {
        if (-not $renameMap.ContainsKey($property.Name)) {
            $newObject | Add-Member -MemberType NoteProperty -Name $property.Name -Value $property.Value
        }
    }
    return $newObject
'@
    $fnEntry = New-Object System.Management.Automation.Runspaces.SessionStateFunctionEntry('Rename-Properties', $renamePropsFunctionBody)
    $iss.Commands.Add($fnEntry)

    $runspacePool = New-ControlledRunspacePool -MinThreads 2 -MaxThreads $script:ControlledParallelismMaxThreads -InitialState $iss
    
    # Define per-workspace metadata collection scriptblock
    $workspacemetadataScript = {
        param(
            $Workspace,
            $DatasetRenameMap,
            $DatasetDatasourceRenameMap,
            $ReportRenameMap,
            $PageRenameMap,
            $DataflowRenameMap,
            $DataflowDatasourceRenameMap,
            $FabricDataflowRenameMap,
            $DataflowLineageRenameMap,
            $GlobalEndpoints,
            $SkipFabricCallsInUSGov,
            $EnvironmentForEndpoints,
            $AccessToken
        )
        
        $workerResults = @{
            WorkspaceName = $Workspace.WorkspaceName
            WorkspaceId = $Workspace.WorkspaceId
            DatasetsInfo = @()
            ReportsInfo = @()
            DatasetSourcesInfo = @()
            DataflowsInfo = @()
            DataflowSourcesInfo = @()
            DatasetNameLookup = @{}
            DataflowNameLookup = @{}
            Success = $true
            ErrorMessage = ''
            DataCount = 0
        }
        
        try {
            $headers = @{
                'Authorization' = "Bearer $AccessToken"
                'Content-Type' = 'application/json'
            }
            
            # Fetch datasets
            $datasetsUrl = "$($GlobalEndpoints.ApiPrefix)/v1.0/myorg/groups/$($Workspace.WorkspaceId)/datasets"
            $datasetsJson = Invoke-RestMethod -Uri $datasetsUrl -Method GET -Headers $headers -ErrorAction Stop 2>$null
            $datasets = @($datasetsJson.value)
            
            foreach ($dataset in $datasets) {
                $renamedDataset = Rename-Properties -object $dataset -renameMap $DatasetRenameMap
                $renamedDataset | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $Workspace.WorkspaceId -Force
                $renamedDataset | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $Workspace.WorkspaceName -Force
                
                $workerResults.DatasetNameLookup[$dataset.id] = $dataset.name
                $workerResults.DatasetsInfo += $renamedDataset
                $workerResults.DataCount++
                
                # Fetch dataset sources
                $datasourcesUrl = "$($GlobalEndpoints.ApiPrefix)/v1.0/myorg/groups/$($Workspace.WorkspaceId)/datasets/$($dataset.id)/datasources"
                try {
                    $datasourcesJson = Invoke-RestMethod -Uri $datasourcesUrl -Method GET -Headers $headers -ErrorAction Stop 2>$null
                    $datasources = @($datasourcesJson.value)
                    
                    foreach ($datasource in $datasources) {
                        $renamedDatasource = Rename-Properties -object $datasource -renameMap $DatasetDatasourceRenameMap
                        $renamedDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $Workspace.WorkspaceId -Force
                        $renamedDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $Workspace.WorkspaceName -Force
                        $renamedDatasource | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.id -Force
                        $renamedDatasource | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.name -Force
                        if ($datasource.connectionDetails) {
                            $renamedDatasource.DatasetDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
                        }
                        $workerResults.DatasetSourcesInfo += $renamedDatasource
                        $workerResults.DataCount++
                    }
                }
                catch { }
            }
            
            # Fetch reports
            $reportsUrl = "$($GlobalEndpoints.ApiPrefix)/v1.0/myorg/groups/$($Workspace.WorkspaceId)/reports"
            $reportsJson = Invoke-RestMethod -Uri $reportsUrl -Method GET -Headers $headers -ErrorAction Stop 2>$null
            $reports = @($reportsJson.value)
            
            foreach ($report in $reports) {
                $renamedReport = Rename-Properties -object $report -renameMap $ReportRenameMap
                $renamedReport | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $Workspace.WorkspaceId -Force
                $renamedReport | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $Workspace.WorkspaceName -Force
                
                $datasetId = $report.datasetId
                if ($datasetId -and $workerResults.DatasetNameLookup.ContainsKey($datasetId)) {
                    $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $workerResults.DatasetNameLookup[$datasetId] -Force
                } else {
                    $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue "Unknown Dataset" -Force
                }
                
                $workerResults.ReportsInfo += $renamedReport
                $workerResults.DataCount++
            }
            
            # Fetch dataflows
            $dataflowsUrl = "$($GlobalEndpoints.ApiPrefix)/v1.0/myorg/groups/$($Workspace.WorkspaceId)/dataflows"
            try {
                $dataflowsJson = Invoke-RestMethod -Uri $dataflowsUrl -Method GET -Headers $headers -ErrorAction Stop 2>$null
                $dataflows = @($dataflowsJson.value)
                
                foreach ($dataflow in $dataflows) {
                    $renamedDataflow = Rename-Properties -object $dataflow -renameMap $DataflowRenameMap
                    $renamedDataflow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $Workspace.WorkspaceId -Force
                    $renamedDataflow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $Workspace.WorkspaceName -Force
                    
                    if ($dataflow.objectId) {
                        $workerResults.DataflowNameLookup[$dataflow.objectId] = $dataflow.name
                    }
                    
                    $workerResults.DataflowsInfo += $renamedDataflow
                    $workerResults.DataCount++
                    
                    # Fetch dataflow sources
                    $dataflowSourcesUrl = "$($GlobalEndpoints.ApiPrefix)/v1.0/myorg/groups/$($Workspace.WorkspaceId)/dataflows/$($dataflow.objectId)/datasources"
                    try {
                        $dataflowSourcesJson = Invoke-RestMethod -Uri $dataflowSourcesUrl -Method GET -Headers $headers -ErrorAction Stop 2>$null
                        $dataflowSources = @($dataflowSourcesJson.value)
                        
                        foreach ($datasource in $dataflowSources) {
                            $renamedDataflowDatasource = Rename-Properties -object $datasource -renameMap $DataflowDatasourceRenameMap
                            $renamedDataflowDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $Workspace.WorkspaceId -Force
                            $renamedDataflowDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $Workspace.WorkspaceName -Force
                            $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowId" -NotePropertyValue $dataflow.objectId -Force
                            if ($datasource.connectionDetails) {
                                $renamedDataflowDatasource.DataflowDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
                            }
                            if ($workerResults.DataflowNameLookup.ContainsKey($dataflow.objectId)) {
                                $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowName" -NotePropertyValue $workerResults.DataflowNameLookup[$dataflow.objectId] -Force
                            }
                            $workerResults.DataflowSourcesInfo += $renamedDataflowDatasource
                            $workerResults.DataCount++
                        }
                    }
                    catch { }
                }
            }
            catch { }
        }
        catch {
            $workerResults.Success = $false
            $workerResults.ErrorMessage = "Worker error: $($_.Exception.Message)"
        }
        
        return $workerResults
    }
    
    # Get current access token for workers
    $currentAccessToken = Get-CurrentAccessToken
    
    # Execute parallel worker jobs
    $workerJobs = @()
    foreach ($workspace in $workspacesInfo) {
        $ps = [System.Management.Automation.PowerShell]::Create()
        $ps.RunspacePool = $runspacePool
        
        $ps.AddScript($workspacemetadataScript) | Out-Null
        $ps.AddArgument($workspace) | Out-Null
        $ps.AddArgument($datasetRenameMap) | Out-Null
        $ps.AddArgument($datasetDatasourceRenameMap) | Out-Null
        $ps.AddArgument($reportRenameMap) | Out-Null
        $ps.AddArgument($pageRenameMap) | Out-Null
        $ps.AddArgument($dataflowRenameMap) | Out-Null
        $ps.AddArgument($dataflowDatasourceRenameMap) | Out-Null
        $ps.AddArgument($fabricDataflowRenameMap) | Out-Null
        $ps.AddArgument($dataflowLineageRenameMap) | Out-Null
        $ps.AddArgument($global:PowerBIEndpoints) | Out-Null
        $ps.AddArgument($script:SkipFabricCallsInUSGov) | Out-Null
        $ps.AddArgument($EnvironmentForEndpoints) | Out-Null
        $ps.AddArgument($currentAccessToken) | Out-Null
        
        $asyncResult = $ps.BeginInvoke()
        $workerJobs += @{
            PowerShell = $ps
            AsyncResult = $asyncResult
            Workspace = $workspace
        }
    }
    
    # Collect results from workers
    Write-Output "[PERF] Collecting metadata from $(@($workerJobs).Count) parallel workspace workers..."
    $parallelErrorCount = 0
    $parallelDataCount = 0
    
    foreach ($job in $workerJobs) {
        try {
            if ($job.AsyncResult.AsyncWaitHandle.WaitOne(120000)) {
                $workerOutput = $job.PowerShell.EndInvoke($job.AsyncResult)
                if ($workerOutput -and $workerOutput.Success) {
                    $datasetsInfo += @($workerOutput.DatasetsInfo)
                    $datasetSourcesInfo += @($workerOutput.DatasetSourcesInfo)
                    $reportsInfo += @($workerOutput.ReportsInfo)
                    $dataflowsInfo += @($workerOutput.DataflowsInfo)
                    $dataflowSourcesInfo += @($workerOutput.DataflowSourcesInfo)
                    
                    foreach ($kvp in $workerOutput.DatasetNameLookup.GetEnumerator()) {
                        $datasetNameLookup[$kvp.Key] = $kvp.Value
                    }
                    foreach ($kvp in $workerOutput.DataflowNameLookup.GetEnumerator()) {
                        $dataflowNameLookup[$kvp.Key] = $kvp.Value
                    }
                    
                    $parallelDataCount += $workerOutput.DataCount
                    Write-Output "Workspace [$($job.Workspace.WorkspaceName)] metadata extraction completed (parallel, $($workerOutput.DataCount) items)."
                }
                else {
                    $parallelErrorCount++
                    Write-Warning "Worker failed for workspace [$($job.Workspace.WorkspaceName)]: $($workerOutput.ErrorMessage)"
                }
            }
            else {
                $parallelErrorCount++
                Write-Warning "Workspace [$($job.Workspace.WorkspaceName)] metadata collection timed out (120s)."
            }
        }
        catch {
            $parallelErrorCount++
            Write-Warning "Error collecting results from workspace [$($job.Workspace.WorkspaceName)]: $_"
        }
        finally {
            $job.PowerShell.Dispose()
        }
    }
    
    $runspacePool.Close()
    $runspacePool.Dispose()
    
    Write-Output "[PERF] Parallel phase 1 metadata collection complete. Items collected: $parallelDataCount, Errors: $parallelErrorCount"
}
else {
    # Sequential fallback
    if ($workspacesInfo.Count -le 1) {
        Write-Output "[PERF] Single workspace or none; skipping parallelism overhead."
    }


# Loop through filtered workspaces (SEQUENTIAL FALLBACK)

    $workspaceDatasets = @($datasets.value)
    if ($script:TargetReportScope.FilterActive -and $script:TargetReportScope.DatasetIds.Count -gt 0) {
        $workspaceDatasets = @($workspaceDatasets | Where-Object { $script:TargetReportScope.DatasetIds -contains [string]$_.id })
    }

    foreach ($dataset in $workspaceDatasets) {
        $renamedDataset = Rename-Properties -object $dataset -renameMap $datasetRenameMap
        $renamedDataset | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
        $renamedDataset | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
        
    # Store the DatasetId and DatasetName in the lookup table
        $datasetNameLookup[$dataset.id] = $dataset.name
        $datasetsInfo += $renamedDataset

        # Fetch dataset sources
        $datasourcesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/datasets/$($dataset.id)/datasources"
        $datasources = Invoke-PowerBIRestMethod -Method GET -Url $datasourcesUrl | ConvertFrom-Json

        foreach ($datasource in $datasources.value) {
            $renamedDatasource = Rename-Properties -object $datasource -renameMap $datasetDatasourceRenameMap
            $renamedDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
            $renamedDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
            $renamedDatasource | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.id -Force
            $renamedDatasource | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.name -Force
            if ($datasource.connectionDetails) {
                $renamedDatasource.DatasetDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
            }
            $datasetSourcesInfo += $renamedDatasource
        }
    }

    # Fetch reports
    $reportsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/reports"
    $reports = Invoke-PowerBIRestMethod -Method GET -Url $reportsUrl | ConvertFrom-Json

    $workspaceReports = @($reports.value)
    if ($script:TargetReportScope.FilterActive -and $script:TargetReportScope.ReportIds.Count -gt 0) {
        $workspaceReports = @($workspaceReports | Where-Object { $script:TargetReportScope.ReportIds -contains [string]$_.id })
    }

	# Create a hash set to store Report IDs
	$reportIds = @{}

    foreach ($report in $workspaceReports) {
        $renamedReport = Rename-Properties -object $report -renameMap $reportRenameMap
        $renamedReport | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
        $renamedReport | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force


        # Retrieve and add the correct DatasetName from the lookup table if DatasetId exists.
        # If the dataset is in a different workspace (cross-workspace dataset), look it up via the API.
        $datasetId = $report.datasetId
        if ($datasetId -and $datasetNameLookup.ContainsKey($datasetId)) {
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $datasetNameLookup[$datasetId] -Force
        } elseif ($datasetId -and
                  -not [string]::IsNullOrWhiteSpace([string]$report.datasetWorkspaceId) -and
                  [string]$report.datasetWorkspaceId -ne $workspace.WorkspaceId) {
            # Dataset lives in a different workspace — fetch its name directly and cache it.
            if (-not $datasetNameLookup.ContainsKey($datasetId)) {
                try {
                    $xwsDatasetUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($report.datasetWorkspaceId)/datasets/$datasetId"
                    $xwsDataset = Invoke-PowerBIRestMethod -Method GET -Url $xwsDatasetUrl | ConvertFrom-Json
                    $datasetNameLookup[$datasetId] = if ($xwsDataset -and $xwsDataset.name) { $xwsDataset.name } else { 'Unknown Dataset' }
                }
                catch {
                    $datasetNameLookup[$datasetId] = 'Unknown Dataset'
                }
            }
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $datasetNameLookup[$datasetId] -Force
        } else {
            $renamedReport | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue "Unknown Dataset" -Force
        }

        $reportsInfo += $renamedReport

        # Fetch report pages
        $pagesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/reports/$($report.id)/pages"
        $pages = Invoke-PowerBIRestMethod -Method GET -Url $pagesUrl | ConvertFrom-Json
        foreach ($page in $pages.value) {
            $renamedPage = Rename-Properties -object $page -renameMap $pageRenameMap
            $renamedPage | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
            $renamedPage | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
            $renamedPage | Add-Member -NotePropertyName "ReportId" -NotePropertyValue $report.id -Force
            $renamedPage | Add-Member -NotePropertyName "ReportName" -NotePropertyValue $report.name -Force
            $reportPagesInfo += $renamedPage

            # Store the report ID in the hash set
            $reportIds[$report.id] = $true
        }
    }
    Export-EnvironmentRows -Rows @($datasetsInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "Datasets"
    Export-EnvironmentRows -Rows @($datasetSourcesInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "DatasetSourcesInfo"
    Export-EnvironmentRows -Rows @($reportsInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "Reports"
    Export-EnvironmentRows -Rows @($reportPagesInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "ReportPages"

Write-Output "Workspace [$($workspace.WorkspaceName)] Report & Model metadata extraction process completed."
}

if ($script:TargetReportScope.FilterActive) {
    if ($script:TargetReportScope.ReportIds.Count -gt 0) {
        $reportsInfo = @($reportsInfo | Where-Object { $script:TargetReportScope.ReportIds -contains [string]$_.ReportId })
        $reportPagesInfo = @($reportPagesInfo | Where-Object { $script:TargetReportScope.ReportIds -contains [string]$_.ReportId })

    $resolvedDatasetIds = @($reportsInfo | ForEach-Object { [string]$_.DatasetId } | Where-Object { -not [string]::IsNullOrWhiteSpace($_) } | Select-Object -Unique)

}

# Fetch Refresh Schedule for Datasets
foreach ($workspace in $workspacesInfo) {

Write-Output "Workspace [$($workspace.WorkspaceName)] Model Refresh Schedule extraction process started."

    foreach ($dataset in $datasetsInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) {
        $refreshScheduleUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/datasets/$($dataset.DatasetId)/refreshSchedule"
        
        try {
            $refreshScheduleRawResponse = Invoke-PowerBIRestMethod -Method GET -Url $refreshScheduleUrl
            if ([string]::IsNullOrWhiteSpace($refreshScheduleRawResponse)) {
                continue
            }

            $refreshScheduleResponse = $refreshScheduleRawResponse | ConvertFrom-Json
            if (-not $refreshScheduleResponse) {
                continue
            }
            
            # Get base properties that apply to all rows
            $renamedScheduleRecord = Rename-Properties -object $refreshScheduleResponse -renameMap $datasetRefreshScheduleRenameMap
            if (-not $renamedScheduleRecord) {
                continue
            }
            
            # Create separate rows for each day-time combination
            $days = if ($refreshScheduleResponse.days) { $refreshScheduleResponse.days } else { @($null) }
            $times = if ($refreshScheduleResponse.times) { $refreshScheduleResponse.times } else { @($null) }
            
            foreach ($day in $days) {
                foreach ($time in $times) {
                    $scheduleRow = $renamedScheduleRecord.PSObject.Copy()
                    $scheduleRow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
                    $scheduleRow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetId" -NotePropertyValue $dataset.DatasetId -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetName" -NotePropertyValue $dataset.DatasetName -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetRefreshScheduleDay" -NotePropertyValue $day -Force
                    $scheduleRow | Add-Member -NotePropertyName "DatasetRefreshScheduleTime" -NotePropertyValue $time -Force
                    
                    $datasetRefreshSchedule += $scheduleRow
                }
            }
        }
        catch {
            # Silently continue if refresh schedule is not available for this dataset
        }
    }
Write-Output "Workspace [$($workspace.WorkspaceName)] Model Refresh Schedule extraction process completed."

    Export-EnvironmentRows -Rows @($datasetRefreshSchedule | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "DatasetRefreshSchedule"
}


# Fetch Dataflows for Workspaces
$fabricGen2SkippedByPolicyLogged = $false
foreach ($workspace in $workspacesInfo) {

Write-Output "Workspace [$($workspace.WorkspaceName)] Dataflow metadata extraction process started."

    # Fetch Gen 1 and Gen 2 dataflows using Power BI API
    $dataflowsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/dataflows"
    $dataflowsResponse = Invoke-PowerBIRestMethod -Method GET -Url $dataflowsUrl | ConvertFrom-Json

    # Ensure response is not null before looping
    if ($dataflowsResponse.value) {
        foreach ($dataflow in $dataflowsResponse.value) {
            $renamedDataflow = Rename-Properties -object $dataflow -renameMap $dataflowRenameMap
            $renamedDataflow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
            $renamedDataflow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force

            # Store DataflowId and DataflowName in a lookup table
            if ($dataflow.objectId) {  
                $dataflowNameLookup[$dataflow.objectId] = $dataflow.name  
            }

            $dataflowsInfo += $renamedDataflow

            # Fetch Dataflow Datasources
            $dataflowSourcesUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$($workspace.WorkspaceId)/dataflows/$($dataflow.objectId)/datasources"
            $dataflowSourcesResponse = Invoke-PowerBIRestMethod -Method GET -Url $dataflowSourcesUrl | ConvertFrom-Json

            # Ensure response is not null before looping
            if ($dataflowSourcesResponse.value) {
                foreach ($datasource in $dataflowSourcesResponse.value) {
                    $renamedDataflowDatasource = Rename-Properties -object $datasource -renameMap $dataflowDatasourceRenameMap

                    # Ensure required fields are included
                    $renamedDataflowDatasource | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
                    $renamedDataflowDatasource | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
                    $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowId" -NotePropertyValue $dataflow.objectId -Force

            		if ($datasource.connectionDetails) {
               		 $renamedDataflowDatasource.DataflowDatasourceConnectionDetails = $datasource.connectionDetails | ConvertTo-Json -Compress
            }
                    

                    if ($dataflowNameLookup.ContainsKey($dataflow.objectId)) {
                        $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowName" -NotePropertyValue $dataflowNameLookup[$dataflow.objectId] -Force
                    } else {
                        $renamedDataflowDatasource | Add-Member -NotePropertyName "DataflowName" -NotePropertyValue "Unknown Dataflow" -Force
                    }

                    # Store in collection
                    $dataflowSourcesInfo += $renamedDataflowDatasource
                }
            }
        }
    }

    # Fetch Gen 2 CICD dataflows using Fabric API.
    # Fabric dataflow metadata endpoint is currently public-cloud only for many tenants.
    $fabricEndpoint = $global:PowerBIEndpoints.FabricApiPrefix
    $fabricGen2Supported = ($fabricEndpoint -match '(?i)^https://api\.fabric\.microsoft\.com$')
    if ($script:SkipFabricCallsInUSGov -and $EnvironmentForEndpoints -match '^(?i)USGov|USGovHigh|USGovMil$') {
        $fabricGen2Supported = $false
        if (-not $fabricGen2SkippedByPolicyLogged) {
            Write-Output "[CONFIG] Fabric Gen2 dataflow metadata calls disabled for government cloud by settings (SkipFabricCallsInUSGov=true)."
            $fabricGen2SkippedByPolicyLogged = $true
        }
    }

    if (-not $fabricGen2Supported) {
        Write-Output "Workspace [$($workspace.WorkspaceName)] Fabric Gen2 dataflow metadata skipped (endpoint not supported in this cloud: $fabricEndpoint)."
    }
    else {
        try {
            $fabricDataflowsUrl = "$fabricEndpoint/v1/workspaces/$($workspace.WorkspaceId)/dataflows"
            $fabricHeaders = @{
                'Authorization' = "Bearer $(Get-CurrentAccessToken)"
                'Content-Type' = 'application/json'
            }
            $fabricDataflowsResponse = Invoke-AuthenticatedRestMethodWithRetry -Uri $fabricDataflowsUrl -Headers $fabricHeaders -Method GET

            # Ensure response is not null before looping
            if ($fabricDataflowsResponse.value) {
                foreach ($fabricDataflow in $fabricDataflowsResponse.value) {
                    # Map Fabric API response to match Power BI API structure using rename map
                    $renamedFabricDataflow = Rename-Properties -object $fabricDataflow -renameMap $fabricDataflowRenameMap
                    $renamedFabricDataflow | Add-Member -NotePropertyName "WorkspaceId" -NotePropertyValue $workspace.WorkspaceId -Force
                    $renamedFabricDataflow | Add-Member -NotePropertyName "WorkspaceName" -NotePropertyValue $workspace.WorkspaceName -Force
                    $renamedFabricDataflow | Add-Member -NotePropertyName "DataflowGeneration" -NotePropertyValue "Gen 2 CICD" -Force

                    # Store DataflowId and DataflowName in a lookup table
                    if ($fabricDataflow.id) {
                        $dataflowNameLookup[$fabricDataflow.id] = $fabricDataflow.displayName
                    }

                    $dataflowsInfo += $renamedFabricDataflow
                }
            }
        }
        catch {
            $fabricErrorSummary = Get-ErrorSummary -ErrorRecord $_
            if ($fabricErrorSummary -match '(?i)FeatureNotAvailable|feature is not available') {
                Write-Output "Workspace [$($workspace.WorkspaceName)] Fabric Gen2 dataflow metadata skipped (FeatureNotAvailable)."
            }
            else {
                Write-Output "Workspace [$($workspace.WorkspaceName)] Fabric Gen2 dataflow metadata skipped due to API error: $fabricErrorSummary"
            }
        }
    }

Write-Output "Workspace [$($workspace.WorkspaceName)] Dataflow metadata extraction process completed."

    Export-EnvironmentRows -Rows @($dataflowsInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "Dataflows"
    Export-EnvironmentRows -Rows @($dataflowSourcesInfo | Where-Object { $_.WorkspaceId -eq $workspace.WorkspaceId }) -WorksheetName "DataflowSourcesInfo"
}
} # end of if/else parallelism block

function Get-ReportQuarantineMatch {
    param(
        [Parameter(Mandatory)]
        [object]$Report,
        [Parameter(Mandatory)]
        [string]$WorkspaceName,
        [Parameter(Mandatory)]
        [object[]]$Entries
    )

    foreach ($entry in $Entries) {
        $idMatch = (-not [string]::IsNullOrWhiteSpace($entry.ReportId)) -and ($entry.ReportId -eq $Report.ReportId)

        $nameMatch = $false
        if (-not [string]::IsNullOrWhiteSpace($entry.ReportName)) {
            $reportNamePattern = $entry.ReportName
            $reportNameUsesWildcard = $reportNamePattern -match '[*?]'
            if ($reportNameUsesWildcard) {
                $nameMatch = $Report.ReportName -like $reportNamePattern
            } else {
                $nameMatch = $reportNamePattern -ieq $Report.ReportName
            }
        }

        $workspaceFilterMatches = $false
        if ([string]::IsNullOrWhiteSpace($entry.WorkspaceName)) {
            $workspaceFilterMatches = $true
        } else {
            $workspacePattern = $entry.WorkspaceName
            $workspaceUsesWildcard = $workspacePattern -match '[*?]'
            if ($workspaceUsesWildcard) {
                $workspaceFilterMatches = $WorkspaceName -like $workspacePattern
            } else {
                $workspaceFilterMatches = $workspacePattern -ieq $WorkspaceName
            }
        }

        if (($idMatch -or $nameMatch) -and $workspaceFilterMatches) {
            return $entry
        }
    }

    return $null
}

function Add-ReportToQuarantine {
    param(
        [Parameter(Mandatory)]
        [string]$CsvPath,
        [Parameter(Mandatory)]
        [string]$ReportId,
        [Parameter(Mandatory)]
        [string]$ReportName,
        [Parameter(Mandatory)]
        [string]$WorkspaceName,
        [Parameter(Mandatory)]
        [string]$Reason
    )

    try {
        $existingRows = @()
        if (Test-Path $CsvPath) {
            $existingRows = Import-Csv -Path $CsvPath
        }

        $alreadyExists = $existingRows | Where-Object {
            (($_.ReportId -eq $ReportId) -or ($_.ReportName -ieq $ReportName)) -and
            ([string]::IsNullOrWhiteSpace($_.WorkspaceName) -or ($_.WorkspaceName -ieq $WorkspaceName)) -and
            ($_.IsActive -match '^(?i:true|1|yes|y)$' -or [string]::IsNullOrWhiteSpace($_.IsActive))
        }

        if ($alreadyExists) {
            return $false
        }

        $newRow = [PSCustomObject]@{
            ReportId = $ReportId
            ReportName = $ReportName
            WorkspaceName = $WorkspaceName
            Reason = $Reason
            IsActive = 'true'
        }

        if ($existingRows -and $existingRows.Count -gt 0) {
            @($existingRows + $newRow) | Export-Csv -Path $CsvPath -NoTypeInformation -Force
        } else {
            $newRow | Export-Csv -Path $CsvPath -NoTypeInformation -Force
        }

        return $true
    }
    catch {
        Write-Output "[WARN] Failed to auto-add report to quarantine list at $CsvPath. Details: $(Get-ErrorSummary -ErrorRecord $_)"
        return $false
    }
}

function Test-ReportIncludedBySettings {
    param(
        [Parameter(Mandatory)]
        [object]$Report,
        [Parameter()]
        [string[]]$IncludeReportIds,
        [Parameter()]
        [string[]]$IncludeReportNames
    )

    $hasIdFilters = $IncludeReportIds -and $IncludeReportIds.Count -gt 0
    $hasNameFilters = $IncludeReportNames -and $IncludeReportNames.Count -gt 0

    if (-not $hasIdFilters -and -not $hasNameFilters) {
        return $true
    }

    if ($hasIdFilters -and ($IncludeReportIds -contains [string]$Report.ReportId)) {
        return $true
    }

    if ($hasNameFilters) {
        foreach ($nameFilter in $IncludeReportNames) {
            $usesWildcard = $nameFilter -match '[*?]'
            if ($usesWildcard) {
                if ([string]$Report.ReportName -like $nameFilter) {
                    return $true
                }
            }
            else {
                if ([string]$Report.ReportName -ieq $nameFilter) {
                    return $true
                }
            }
        }
    }

    return $false
}

# Function to Export Using REST API and Stored Token
function Export-ReportUsingAPI {
    param (
        [string]$GroupId,
        [string]$ReportId,
        [string]$OutFilePath,
        [string]$DownloadType
    )

    $accessToken = Get-CurrentAccessToken
    $apiUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$GroupId/reports/$ReportId/Export?downloadType=$DownloadType"

    $downloadSucceeded = $false
    $failureReason = $null
    $script:LastAuthenticatedRestErrorSummary = $null

    try {
        $null = Invoke-AuthenticatedRestMethodWithRetry -Uri $apiUrl `
            -Headers @{ Authorization = "Bearer $accessToken" } `
            -Method GET `
            -OutFile $OutFilePath `
            -ErrorAction Stop

        if (Test-Path $OutFilePath) {
            $fileInfo = Get-Item $OutFilePath -ErrorAction SilentlyContinue
            if ($fileInfo -and $fileInfo.Length -gt 0) {
                $downloadSucceeded = $true
            } else {
                $failureReason = "Export created an empty file."
            }
        } else {
            if (-not [string]::IsNullOrWhiteSpace($script:LastAuthenticatedRestErrorSummary)) {
                $failureReason = "Export API call did not produce a file. Last API error: $script:LastAuthenticatedRestErrorSummary"
            } else {
                $failureReason = "Export API call did not produce a file."
            }
        }
    } catch {
        $failureReason = Get-ErrorSummary -ErrorRecord $_
    }

    return [PSCustomObject]@{
        Success = $downloadSucceeded
        ErrorMessage = $failureReason
    }
}

    $elapsed      = 0
    $sasUrl       = $null
    $retry409Count = 0

    try {
        while ($true) {
            $accessToken = Get-CurrentAccessToken
            $headers = @{
                Authorization = "Bearer $accessToken"
                Accept        = 'application/json, text/plain, */*'
                Origin        = $global:PowerBIEndpoints.WebPrefix
                Referer       = "$($global:PowerBIEndpoints.WebPrefix)/"
            }

            $response = $null
            try {
                $response = Invoke-RestMethod -Method GET -Uri $exportUrl -Headers $headers -ErrorAction Stop
            }
            catch {
                $httpStatus = $null
                try { $httpStatus = [int]$_.Exception.Response.StatusCode } catch {}

                if ($httpStatus -eq 409 -and $retry409Count -lt $Max409Retries) {
                    $retry409Count++
                    $waitSeconds = [Math]::Min(20, ($retry409Count * 5))
                    Write-Output "[INFO] Web UI export endpoint returned HTTP 409 conflict for report $ReportId. Retry $retry409Count/$Max409Retries in $waitSeconds seconds..."
                    Start-Sleep -Seconds $waitSeconds
                    $elapsed += $waitSeconds
                    continue
                }

                $detail = Get-ErrorSummary -ErrorRecord $_
                return [PSCustomObject]@{
                    Success      = $false
                    ErrorMessage = "Web UI export cluster call failed$(if ($httpStatus) { " (HTTP $httpStatus)" }): $detail"
                }
            }

            if ($response.url) {
                $sasUrl = $response.url
                break
            }

            $elapsed += $pollInterval
            if ($elapsed -ge $MaxWaitSeconds) {
                return [PSCustomObject]@{
                    Success      = $false
                    ErrorMessage = "Web UI export timed out after $MaxWaitSeconds seconds waiting for SAS URL. Last response status: $($response.status)"
                }
            }
            Start-Sleep -Seconds $pollInterval
        }

        Invoke-WebRequest -Uri $sasUrl -OutFile $OutFilePath -Method GET -UseBasicParsing -ErrorAction Stop
        if (Test-Path $OutFilePath) {
            $fileInfo = Get-Item $OutFilePath -ErrorAction SilentlyContinue
            if ($fileInfo -and $fileInfo.Length -gt 0) {
                return [PSCustomObject]@{ Success = $true; ErrorMessage = $null }
            }
        }
        return [PSCustomObject]@{
            Success      = $false
            ErrorMessage = 'Web UI export SAS download produced an empty or missing file.'
        }
    }
    catch {
        return [PSCustomObject]@{
            Success      = $false
            ErrorMessage = "Web UI export failed: $(Get-ErrorSummary -ErrorRecord $_)"
        }
    }
}

function Resolve-WabiClusterHostForReport {
    param(
        [Parameter(Mandatory)]
        [string]$ReportWebUrl,
        [string]$WorkspaceId,
        [string]$ReportId
    )

    $candidates = @()
    if (-not [string]::IsNullOrWhiteSpace($ReportWebUrl)) {
        $candidates += $ReportWebUrl
    }
    if (-not [string]::IsNullOrWhiteSpace($WorkspaceId) -and -not [string]::IsNullOrWhiteSpace($ReportId)) {
        $candidates += "$($global:PowerBIEndpoints.WebPrefix)/groups/$WorkspaceId/reports/$ReportId"
    }

    foreach ($candidateUrl in ($candidates | Select-Object -Unique)) {
        try {
            $accessToken = Get-CurrentAccessToken
            $headers = @{
                Authorization = "Bearer $accessToken"
                Accept        = 'text/html,application/xhtml+xml,application/xml;q=0.9,*/*;q=0.8'
                Referer       = "$($global:PowerBIEndpoints.WebPrefix)/"
            }

            $response = $null
            try {
                $response = Invoke-WebRequest -Uri $candidateUrl -Method Head -Headers $headers -MaximumRedirection 0 -UseBasicParsing -ErrorAction Stop
            }
            catch {
                $response = $_.Exception.Response
            }

            if ($response) {
                $location = $null
                try { $location = $response.Headers['Location'] } catch {}

                if (-not [string]::IsNullOrWhiteSpace($location)) {
                    $baseUri = [Uri]$candidateUrl
                    $locationUri = $null

                    if ([Uri]::TryCreate($location, [System.UriKind]::Absolute, [ref]$locationUri)) {
                        if ($locationUri.Host -match '(?i)^wabi-.*\.analysis\.') {
                            return $locationUri.Host
                        }
                    }
                    elseif ([Uri]::TryCreate($baseUri, $location, [ref]$locationUri)) {
                        if ($locationUri.Host -match '(?i)^wabi-.*\.analysis\.') {
                            return $locationUri.Host
                        }
                    }
                }

                try {
                    if ($response.ResponseUri -and $response.ResponseUri.Host -match '(?i)^wabi-.*\.analysis\.') {
                        return $response.ResponseUri.Host
                    }
                }
                catch {}
            }
        }
        catch {}
    }

    return $null
}

	# Function: get first free drive letter (Z down to D)
	function Get-FreeDriveLetter {
	    $used = Get-PSDrive -PSProvider FileSystem | ForEach-Object { $_.Name }
	    foreach ($letter in [char[]](90..68)) {  # Z..Y..X.. down to D
	        if ($used -notcontains $letter) { return "${letter}:" }
	    }
	    throw "No free drive letters available."
	}

    function Invoke-PbiToolsProcess {
        param(
            [Parameter(Mandatory)]
            [string[]]$Arguments,
            [Parameter(Mandatory)]
            [string]$OperationName
        )

        $pbiToolsLogFolder = Join-Path -Path $configFolder -ChildPath "Logs\PbiTools"
        if (-not (Test-Path $pbiToolsLogFolder)) {
            New-Item -Path $pbiToolsLogFolder -ItemType Directory -Force | Out-Null
        }

        $runId = [Guid]::NewGuid().ToString('N')
        $stdoutPath = Join-Path -Path $pbiToolsLogFolder -ChildPath ("{0}_{1}_stdout.log" -f $OperationName, $runId)
        $stderrPath = Join-Path -Path $pbiToolsLogFolder -ChildPath ("{0}_{1}_stderr.log" -f $OperationName, $runId)

        $allOutput = & $pbiToolsPath @Arguments 2>&1
        $exitCode = $LASTEXITCODE

        $stderrLines = @($allOutput | Where-Object { $_ -is [System.Management.Automation.ErrorRecord] } | ForEach-Object { $_.ToString() })
        $stdoutLines = @($allOutput | Where-Object { $_ -isnot [System.Management.Automation.ErrorRecord] } | ForEach-Object { [string]$_ })

        $stdoutLines | Set-Content -Path $stdoutPath -Encoding UTF8
        $stderrLines | Set-Content -Path $stderrPath -Encoding UTF8

        $stdoutText = if (Test-Path $stdoutPath) { Get-Content -Path $stdoutPath -Raw -ErrorAction SilentlyContinue } else { '' }
        $stderrText = if (Test-Path $stderrPath) { Get-Content -Path $stderrPath -Raw -ErrorAction SilentlyContinue } else { '' }

        return [PSCustomObject]@{
            ExitCode = [int]$exitCode
            StdOut = [string]$stdoutText
            StdErr = [string]$stderrText
            StdOutPath = $stdoutPath
            StdErrPath = $stderrPath
        }
    }

    function Get-PbiToolsFailureSummary {
        param(
            [Parameter(Mandatory)]
            [object]$ProcessResult
        )

        $combinedText = @($ProcessResult.StdErr, $ProcessResult.StdOut) -join "`n"
        $combinedLines = @($combinedText -split "`r?`n" | Where-Object { -not [string]::IsNullOrWhiteSpace($_) })
        $noiseLinePatterns = @(
            '(?i)^pbi-tools \| copyright',
            '(?i)^this program is free software',
            '(?i)^you can redistribute it and/or modify',
            '(?i)^it under the terms of the gnu affero',
            '(?i)^by the free software foundation',
            '(?i)^this program is distributed in the hope',
            '(?i)^but without any warranty',
            '(?i)^MERCHANTABILITY or FITNESS FOR A PARTICULAR PURPOSE',
            '(?i)^See <https://goto\.pbi\.tools/license>',
            '(?i)^Reading PBIXPROJ settings from:'
        )

        $signalLines = @($combinedLines | Where-Object {
            $line = [string]$_
            -not ($noiseLinePatterns | Where-Object { $line -match $_ })
        })

        $priorityLines = @($signalLines | Where-Object {
            $_ -match '(?i)error|exception|failed|could not|not be exported|invalid|denied|not found|timeout'
        })

        $tailLines = if ($priorityLines.Count -gt 0) { @($priorityLines | Select-Object -Last 6) } else { @($signalLines | Select-Object -Last 6) }
        $tailText = ($tailLines -join ' | ')

        if ($tailText.Length -gt 700) {
            $tailText = $tailText.Substring(0, 700) + '...'
        }

        if ($combinedText -match '(?i)member .* is already present') {
            if ([string]::IsNullOrWhiteSpace($tailText)) {
                $tailText = 'Observed duplicate TypeData member message while running pbi-tools.'
            }
            else {
                $tailText = "$tailText | Observed duplicate TypeData member message while running pbi-tools."
            }
        }

        if ($combinedText -match '(?i)PathTooLongException|DirectoryNotFoundException|The specified path, file name, or both are too long|Could not find a part of the path') {
            $longPathGuidance = 'Detected path-length/report-serialization failure. Recommended bypass: enable Windows long paths (LongPathsEnabled=1) and rerun.'
            if ([string]::IsNullOrWhiteSpace($tailText)) {
                $tailText = $longPathGuidance
            }
            else {
                $tailText = "$tailText | $longPathGuidance"
            }
        }

        $logPointer = "stdout=$($ProcessResult.StdOutPath); stderr=$($ProcessResult.StdErrPath)"
        if ([string]::IsNullOrWhiteSpace($tailText)) {
            return "Diagnostic logs: $logPointer"
        }

        return "Details: $tailText. Diagnostic logs: $logPointer"
    }

    $reportExportFailures = @()
    $modelExtractionFailures = @()

    $includeReportIds = @($script:ImpactIQSettings.IncludeReportIds | ForEach-Object { [string]$_ } | Where-Object { -not [string]::IsNullOrWhiteSpace($_) })
    $includeReportNames = @($script:ImpactIQSettings.IncludeReportNames | ForEach-Object { [string]$_ } | Where-Object { -not [string]::IsNullOrWhiteSpace($_) })
    $reportFilterIsActive = ($includeReportIds.Count -gt 0 -or $includeReportNames.Count -gt 0)
    if ($reportFilterIsActive) {
        Write-Output "[CONFIG] Report include filter active. IncludeReportIds=$($includeReportIds.Count), IncludeReportNames=$($includeReportNames.Count). Only matching reports will be processed."
    }

	foreach ($workspace in $workspacesInfo) {
	    $workspaceName = $workspace.WorkspaceName
	    $workspaceId = $workspace.WorkspaceId
	
	    # Clean up workspace name
	    $cleanWorkspaceName = $workspaceName -replace '\[', '(' -replace '\]', ')'
	    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
	    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
	
	    # Fetch reports from the existing list
	    $reports = $reportsInfo | Where-Object { $_.WorkspaceId -eq $workspaceId }
	
	    foreach ($report in $reports) {
	        $reportName = $report.ReportName
	        $reportId = $report.ReportId
        $reportType = [string]$report.ReportType

        if ($reportFilterIsActive -and -not (Test-ReportIncludedBySettings -Report $report -IncludeReportIds $includeReportIds -IncludeReportNames $includeReportNames)) {
            continue
        }

        # Clean up report name
        $cleanReportName = $reportName -replace '\[', '(' -replace '\]', ')'
        $cleanReportName = $cleanReportName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
        $cleanReportName = $cleanReportName.TrimStart()

        # Power BI API returns reportType=PowerBIReport for Scorecards, so also detect via embedUrl.
        $isScorecard = ([string]$report.ReportEmbedUrl -match '(?i)scorecardEmbed')
        $exportableReportTypes = @('PowerBIReport', 'PaginatedReport')
        if ($isScorecard -or ($reportType -and ($exportableReportTypes -notcontains $reportType))) {
            $detectedType = if ($isScorecard) { "Scorecard (detected via embedUrl)" } else { $reportType }
            $skipReason = "Non-exportable item type for backup flow: $detectedType"
            Write-Output "[INFO] Skipping $cleanWorkspaceName ~ $cleanReportName. Reason: $skipReason"
            continue
        }
	        $fileExtension = if ($report.WebUrl -like "*/rdlreports/*") { "rdl" } else { "pbix" }
	        $filename = "$cleanWorkspaceName ~ $cleanReportName.$fileExtension"
	        $filepath = Join-Path -Path $newDateFolder -ChildPath $filename
	        $extractFolder = Join-Path -Path $tempExtractFolder -ChildPath "$cleanWorkspaceName ~ $cleanReportName"

            $quarantineMatch = Get-ReportQuarantineMatch -Report $report -WorkspaceName $workspaceName -Entries $reportQuarantineEntries
            if ($quarantineMatch) {
                $quarantineReason = if ([string]::IsNullOrWhiteSpace($quarantineMatch.Reason)) { 'No reason provided.' } else { $quarantineMatch.Reason }
                Write-Output "[QUARANTINE] Skipping report export for $cleanWorkspaceName ~ $cleanReportName. Reason: $quarantineReason"
                continue
            }

            # Skip reports not created from a PBIX file — these cannot be exported.
            # ReportIsFromPbix=False typically means the report was published as dataset-only
            # or created through the service; the PBIX does not exist (ExportPBIX_ModelessWorkbookNotFound).
            if ([string]$report.ReportIsFromPbix -match '^(?i:false|0)$') {
                $skipReason = 'Report was not created from a PBIX file (ReportIsFromPbix=False). PBIX download is unavailable for this report type.'
                Write-Output "[INFO] Skipping $cleanWorkspaceName ~ $cleanReportName. Reason: $skipReason"
                $reportExportFailures += [PSCustomObject]@{
                    WorkspaceId           = $workspaceId
                    WorkspaceName         = $workspaceName
                    ReportId              = $reportId
                    ReportName            = $reportName
                    ReportWebUrl          = $report.ReportWebUrl
                    ReportType            = $report.ReportType
                    DownloadTypeAttempted = 'NotApplicable'
                    FailureReason         = $skipReason
                    FailureTimestamp      = (Get-Date -Format o)
                }
                continue
            }

            # Resume/Append: skip download + extraction if this file already exists and run is non-destructive
            if (($resumeMode -or -not $script:OverwriteOutputs) -and (Test-Path $filepath)) {
                $skipTag = if ($resumeMode) { '[RESUME]' } else { '[APPEND]' }
                Write-Output "$skipTag Skipping already-backed-up report: $cleanWorkspaceName ~ $cleanReportName"
                continue
            }

            if ($script:EnableReportChangeDetection) {
                $stateKey = "$workspaceId|$reportId"
                $currentFingerprint = New-ReportFingerprint -Report $report -WorkspaceId $workspaceId -WorkspaceIsOnDedicatedCapacity ([bool]$workspace.WorkspaceIsOnDedicatedCapacity)
                $previousState = $null
                if ($script:ReportExportState.ContainsKey($stateKey)) {
                    $previousState = $script:ReportExportState[$stateKey]
                }

                if ($previousState -and ([string]$previousState.Fingerprint -eq $currentFingerprint) -and (Test-Path $filepath)) {
                    Write-Output "[STATE] Skipping unchanged report export: $cleanWorkspaceName ~ $cleanReportName"
                    continue
                }
            }

	        # Delete file if it already exists
	        if (Test-Path $filepath) {
	            Remove-Item $filepath -Force
	        }
	
	        Write-Host "Exporting $cleanWorkspaceName ~ $cleanReportName"
	
            # Export based on Workspace Type
            $downloadResult = $null
            $initialDownloadType = $null
            try {
                if ($workspace.WorkspaceIsOnDedicatedCapacity -eq $false) {
                    # Pro Workspace → IncludeModel
                    $initialDownloadType = "IncludeModel"
                    $downloadResult = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType $initialDownloadType
                } else {
                    # Premium/Fabric Workspace → LiveConnect
                    $initialDownloadType = "LiveConnect"
                    $downloadResult = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType $initialDownloadType
                }
            }
            catch {
                $downloadResult = [PSCustomObject]@{
                    Success = $false
                    ErrorMessage = Get-ErrorSummary -ErrorRecord $_
                }
            }

            $usedDownloadType = $initialDownloadType

            # Fallback: on auth/permission-style failures for IncludeModel, retry once with LiveConnect.
            if ((-not $downloadResult -or -not $downloadResult.Success) -and $initialDownloadType -eq "IncludeModel") {
                $fallbackReason = if ($downloadResult -and $downloadResult.ErrorMessage) { $downloadResult.ErrorMessage } else { "Unknown export failure." }
                if ($fallbackReason -match '(?i)\b401\b|\b403\b|unauthorized|forbidden|modelexportactiondenied|exportactiondenied') {
                    Write-Output "[INFO] Export retry with LiveConnect for $cleanWorkspaceName ~ $cleanReportName after IncludeModel failure: $fallbackReason"
                    $downloadResult = Export-ReportUsingAPI -GroupId $workspaceId -ReportId $reportId -OutFilePath $filepath -DownloadType "LiveConnect"
                    $usedDownloadType = "LiveConnect"
                }
            }

            # Fallback: for PremiumFiles / large model storage format errors, retry using the web UI
            # cluster download method. This succeeds where the standard Export API returns HTTP 400
            # with ServerError_PremiumFilesErrors_OperationIsNotSupportedForPremiumFilesModel.
            $clusterHostForReport = $null
            $clusterCacheKey = "$workspaceId|$reportId"
            if ($script:WabiClusterHostCache.ContainsKey($clusterCacheKey)) {
                $clusterHostForReport = [string]$script:WabiClusterHostCache[$clusterCacheKey]
            }
            else {
                $clusterHostForReport = Resolve-WabiClusterHostForReport -ReportWebUrl ([string]$report.ReportWebUrl) -WorkspaceId $workspaceId -ReportId $reportId
                if ([string]::IsNullOrWhiteSpace($clusterHostForReport) -and -not [string]::IsNullOrWhiteSpace($global:PowerBIEndpoints.WabiClusterHost)) {
                    $clusterHostForReport = $global:PowerBIEndpoints.WabiClusterHost
                }
                if (-not [string]::IsNullOrWhiteSpace($clusterHostForReport)) {
                    $script:WabiClusterHostCache[$clusterCacheKey] = $clusterHostForReport
                }
            }

            if ((-not $downloadResult -or -not $downloadResult.Success) -and
                -not [string]::IsNullOrWhiteSpace($clusterHostForReport)) {
                $fallbackReason = if ($downloadResult -and $downloadResult.ErrorMessage) { $downloadResult.ErrorMessage } else { 'Unknown export failure.' }
                if ($fallbackReason -match '(?i)PremiumFiles|OperationIsNotSupportedForPremiumFiles|\b400\b|Bad Request') {
                    Write-Output "[INFO] Export retry using web UI method for $cleanWorkspaceName ~ $cleanReportName via cluster host '$clusterHostForReport' after API failure: $fallbackReason"
                    $downloadResult = Export-ReportUsingWebUIMethod -ReportId $reportId -ClusterHost $clusterHostForReport -OutFilePath $filepath
                    $usedDownloadType = if ($downloadResult.Success) { 'WebUI' } else { 'WebUI(failed)' }
                }
            }

            if (-not $downloadResult -or -not $downloadResult.Success) {
                $exportFailureReason = if ($downloadResult -and $downloadResult.ErrorMessage) { $downloadResult.ErrorMessage } else { 'Unknown export failure.' }
                Write-Output "[WARN] Export skipped for $cleanWorkspaceName ~ $cleanReportName. Reason: $exportFailureReason"

                if ($exportFailureReason -match '(?i)\b401\b|\b403\b|unauthorized|forbidden|modelexportactiondenied|exportactiondenied') {
                    $autoQuarantineReason = "Auto-added by script after export-denied/auth failure: $exportFailureReason"
                    $wasAdded = Add-ReportToQuarantine -CsvPath $reportQuarantineListPath `
                                                      -ReportId $reportId `
                                                      -ReportName $reportName `
                                                      -WorkspaceName $workspaceName `
                                                      -Reason $autoQuarantineReason
                    if ($wasAdded) {
                        Write-Output "[QUARANTINE] Auto-added report to quarantine list: $cleanWorkspaceName ~ $cleanReportName"
                        $reportQuarantineEntries += [PSCustomObject]@{
                            ReportId = $reportId
                            ReportName = $reportName
                            WorkspaceName = $workspaceName
                            Reason = $autoQuarantineReason
                            IsActive = 'true'
                        }
                    }
                }

                $reportExportFailures += [PSCustomObject]@{
                    WorkspaceId = $workspaceId
                    WorkspaceName = $workspaceName
                    ReportId = $reportId
                    ReportName = $reportName
                    ReportWebUrl = $report.ReportWebUrl
                    ReportType = $report.ReportType
                    DownloadTypeAttempted = $usedDownloadType
                    FailureReason = $exportFailureReason
                    FailureTimestamp = (Get-Date -Format o)
                }
                continue
            }

            Write-Output "[INFO] Exported report file: $cleanWorkspaceName ~ $cleanReportName (DownloadType: $usedDownloadType)"

            if ($script:EnableReportChangeDetection) {
                $stateKey = "$workspaceId|$reportId"
                $stateFingerprint = New-ReportFingerprint -Report $report -WorkspaceId $workspaceId -WorkspaceIsOnDedicatedCapacity ([bool]$workspace.WorkspaceIsOnDedicatedCapacity)
                $script:ReportExportState[$stateKey] = [PSCustomObject]@{
                    Key = $stateKey
                    WorkspaceId = [string]$workspaceId
                    WorkspaceName = [string]$workspaceName
                    ReportId = [string]$reportId
                    ReportName = [string]$reportName
                    FilePath = [string]$filepath
                    Fingerprint = [string]$stateFingerprint
                    LastExportUtc = (Get-Date).ToUniversalTime().ToString('o')
                }
                $script:ReportExportStateDirty = $true
            }
			
        # Only process model extraction for Pro Workspaces
        if ($workspace.WorkspaceIsOnDedicatedCapacity -eq $false) {
            if (Test-IsKnownNoModelReport -ReportId ([string]$reportId) -ReportName ([string]$reportName)) {
                $knownSkipReason = 'Report is on KnownNoModel skip list (configured or auto-learned).'
                Write-Output "[INFO] Model extraction skipped for $cleanWorkspaceName ~ $cleanReportName. Reason: $knownSkipReason"
                $modelExtractionFailures += [PSCustomObject]@{
                    WorkspaceId = $workspaceId
                    WorkspaceName = $workspaceName
                    ReportId = $reportId
                    ReportName = $reportName
                    DownloadTypeAttempted = $usedDownloadType
                    FailureReason = $knownSkipReason
                    IsExpected = $true
                    FailureTimestamp = (Get-Date -Format o)
                }
                continue
            }

            Write-Host "Extracting $cleanWorkspaceName ~ $cleanReportName"
		
		    $driveLetter = Get-FreeDriveLetter
		    $quotedExtractFolder = "`"$extractFolder`""
		
		    # ✅ Make sure the extract folder exists before subst
		    if (-not (Test-Path $extractFolder)) {
		        New-Item -ItemType Directory -Path $extractFolder -Force | Out-Null
		    }
		
		    if (Test-Path $driveLetter) { subst $driveLetter /D }
		
		    $substCommand = "subst $driveLetter $quotedExtractFolder"
		    cmd.exe /c $substCommand
		
		    if (-not (Test-Path $driveLetter)) {
                Write-Output "[INFO] Model was NOT extracted for $cleanWorkspaceName ~ $cleanReportName. Reason: Failed to map extraction folder to temporary drive letter ($driveLetter)."
		        continue
		    }
		
		    try {
                $modelExtracted = $false
                $modelFailureReason = $null
                $isExpectedNoEmbeddedModel = $false
                $attemptPlans = @(
                    [PSCustomObject]@{ Name = 'subst-raw'; ExtractFolderArg = $driveLetter; UseRawSerialization = $true },
                    [PSCustomObject]@{ Name = 'direct-default'; ExtractFolderArg = $extractFolder; UseRawSerialization = $false }
                )
                $maxModelExtractionAttempts = $attemptPlans.Count

                for ($attempt = 1; $attempt -le $maxModelExtractionAttempts; $attempt++) {
                    $attemptPlan = $attemptPlans[$attempt - 1]
                    Write-Host "Attempting model extraction from $cleanWorkspaceName ~ $cleanReportName (attempt $attempt/$maxModelExtractionAttempts, strategy: $($attemptPlan.Name))"

                    $extractArgs = @('extract', $filepath, '-extractFolder', $attemptPlan.ExtractFolderArg)
                    if ($attemptPlan.UseRawSerialization) {
                        $extractArgs += @('-modelSerialization', 'Raw')
                    }

                    $extractResult = Invoke-PbiToolsProcess -Arguments $extractArgs -OperationName 'extract'
                    $extractExitCode = $extractResult.ExitCode

                    if ($extractExitCode -ne 0) {
                        $extractSummary = Get-PbiToolsFailureSummary -ProcessResult $extractResult
                        $modelFailureReason = "pbi-tools extract returned exit code $extractExitCode. $extractSummary"
                        if ($attempt -lt $maxModelExtractionAttempts) {
                            Write-Output "[WARN] Model extraction attempt failed for $cleanWorkspaceName ~ $cleanReportName ($modelFailureReason) Retrying once..."
                            Start-Sleep -Seconds 2
                            continue
                        }
                        break
                    }

                    $generateArgs = @('generate-bim', $attemptPlan.ExtractFolderArg, '-transforms', 'RemovePBIDataSourceVersion')
                    $generateResult = Invoke-PbiToolsProcess -Arguments $generateArgs -OperationName 'generate-bim'
                    $generateBimExitCode = $generateResult.ExitCode

                    if ($generateBimExitCode -ne 0) {
                        $generateSummary = Get-PbiToolsFailureSummary -ProcessResult $generateResult
                        if ($generateBimExitCode -eq -8) {
                            $isExpectedNoEmbeddedModel = $true
                            $modelFailureReason = "pbi-tools generate-bim returned exit code -8 (no embedded model; typically live connection, scorecard, or service-authored report). $generateSummary"
                            $wasAddedToNoModel = Add-KnownNoModelReport -ReportId ([string]$reportId) -ReportName ([string]$reportName)
                            if ($wasAddedToNoModel) {
                                Write-Output "[INFO] Added report to KnownNoModel skip list: $cleanWorkspaceName ~ $cleanReportName"
                            }
                            break
                        }

                        $modelFailureReason = "pbi-tools generate-bim returned exit code $generateBimExitCode. $generateSummary"
                        if ($attempt -lt $maxModelExtractionAttempts) {
                            Write-Output "[WARN] Model generate-bim attempt failed for $cleanWorkspaceName ~ $cleanReportName ($modelFailureReason) Retrying once..."
                            Start-Sleep -Seconds 2
                            continue
                        }
                        break
                    }

                    $bimFilesGenerated = Get-ChildItem -Path $extractFolder -Filter *.bim
                    if (-not $bimFilesGenerated -or $bimFilesGenerated.Count -eq 0) {
                        $isExpectedNoEmbeddedModel = $true
                        $modelFailureReason = "No .bim file was generated (likely live connection or no embedded model)."
                        break
                    }

                    foreach ($bimFile in $bimFilesGenerated) {
                        $newBimName = "$cleanWorkspaceName ~ $cleanReportName.bim"
                        $newBimPath = Join-Path -Path $tempExtractFolder -ChildPath $newBimName

                        Move-Item -Path $bimFile.FullName -Destination $newBimPath -Force
                    }

                    $modelExtracted = $true
                    break
                }

                if ($modelExtracted) {
                    Write-Output "[INFO] Model extracted successfully for $cleanWorkspaceName ~ $cleanReportName."
                }
                else {
                    if ([string]::IsNullOrWhiteSpace($modelFailureReason)) {
                        $modelFailureReason = 'Unknown model extraction failure.'
                    }

                    if ($isExpectedNoEmbeddedModel) {
                        Write-Output "[INFO] Model extraction skipped for $cleanWorkspaceName ~ $cleanReportName. Reason: $modelFailureReason"
                    }
                    else {
                        Write-Output "[WARN] Model extraction failed for $cleanWorkspaceName ~ $cleanReportName. Reason: $modelFailureReason"
                    }

                    $modelExtractionFailures += [PSCustomObject]@{
                        WorkspaceId = $workspaceId
                        WorkspaceName = $workspaceName
                        ReportId = $reportId
                        ReportName = $reportName
                        DownloadTypeAttempted = $usedDownloadType
                        FailureReason = $modelFailureReason
                        IsExpected = $isExpectedNoEmbeddedModel
                        FailureTimestamp = (Get-Date -Format o)
                    }
                }
            }
            catch {

                $exceptionReason = Get-ErrorSummary -ErrorRecord $_
                Write-Output "[WARN] Model extraction failed for $cleanWorkspaceName ~ $cleanReportName. Reason: $exceptionReason"

                    WorkspaceId = $workspaceId

                    ReportId = $reportId
                    ReportName = $reportName
                    DownloadTypeAttempted = $usedDownloadType
                    FailureReason = $exceptionReason
                    IsExpected = $false
                    FailureTimestamp = (Get-Date -Format o)
                }
		    }
		    finally {
		        # ✅ Always clean up subst drive, even if extract/generate fails
		        subst $driveLetter /D 2>$null
		    }
        }
        else {
            Write-Output "[INFO] Model extraction skipped for $cleanWorkspaceName ~ $cleanReportName. Reason: Workspace is Premium/Fabric (LiveConnect export)."
        }
	}
}

function Normalize-ImpactIQNameForMatch {
    param(
        [Parameter()]
        [string]$Name
    )

    if ([string]::IsNullOrWhiteSpace($Name)) {
        return ''
    }

    $normalized = $Name.ToLowerInvariant()
    $normalized = $normalized -replace '[^a-z0-9 ]', ' '
    $normalized = $normalized -replace '\b(report|model|dashboard)\b', ' '
    $normalized = $normalized -replace '\s+', ' '
    return $normalized.Trim()
}

function Test-IsExpectedMissingBim {
    param(
        [Parameter(Mandatory)]
        [string]$WorkspaceName,
        [Parameter(Mandatory)]
        [string]$DatasetName,
        [Parameter()]
        [object[]]$QuarantineEntries,
        [Parameter()]
        [object[]]$ReportFailures
    )

    $datasetNorm = Normalize-ImpactIQNameForMatch -Name $DatasetName
    if ([string]::IsNullOrWhiteSpace($datasetNorm)) {
        return $false
    }

    foreach ($entry in @($QuarantineEntries)) {
        if ([string]::IsNullOrWhiteSpace([string]$entry.WorkspaceName) -or ([string]$entry.WorkspaceName -ieq $WorkspaceName)) {
            $entryNorm = Normalize-ImpactIQNameForMatch -Name ([string]$entry.ReportName)
            if (-not [string]::IsNullOrWhiteSpace($entryNorm) -and ($datasetNorm.Contains($entryNorm) -or $entryNorm.Contains($datasetNorm))) {
                return $true
            }
        }
    }

    foreach ($failure in @($ReportFailures)) {
        if ([string]$failure.WorkspaceName -ieq $WorkspaceName) {
            $reason = [string]$failure.FailureReason
            if ($reason -match '(?i)Report was not created from a PBIX|^Quarantine:') {
                $failureNorm = Normalize-ImpactIQNameForMatch -Name ([string]$failure.ReportName)
                if (-not [string]::IsNullOrWhiteSpace($failureNorm) -and ($datasetNorm.Contains($failureNorm) -or $failureNorm.Contains($datasetNorm))) {
                    return $true
                }
            }
        }
    }

    return $false
}

# Function to download and extract Fabric (Gen2) dataflows
function Export-FabricDataflow {
    param (
        [string]$WorkspaceId,
        [string]$DataflowId,
        [string]$WorkspaceName,
        [string]$DataflowName,
        [string]$OutputPath,
        [string]$AccessToken
    )
    
    # Clean up names
    $cleanWorkspaceName = $WorkspaceName -replace '\[', '(' -replace '\]', ')'
    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
    
    $cleanDataflowName = $DataflowName -replace '\[', '(' -replace '\]', ')'
    $cleanDataflowName = $cleanDataflowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanDataflowName = $cleanDataflowName.TrimStart()
    
    $fabricApiBaseUri = "$($global:PowerBIEndpoints.FabricApiPrefix)/v1/workspaces"
    $endpoint = "$fabricApiBaseUri/$WorkspaceId/dataflows/$DataflowId/getDefinition"
    
    $headers = @{
        "Authorization" = "Bearer $AccessToken"
        "Content-Type"  = "application/json"
    }
    
    try {
        $response = Invoke-AuthenticatedRestMethodWithRetry -Uri $endpoint -Method POST -Headers $headers -Body "{}"
        
        if (-not $response.definition.parts) {
            return $null
        }
        
        # Create temp directory for extraction
        $tempDir = Join-Path $env:TEMP "FabricDataflow_$DataflowId"
        if (Test-Path $tempDir) {
            Remove-Item -Path $tempDir -Recurse -Force
        }
        New-Item -Path $tempDir -ItemType Directory -Force | Out-Null
        
        # Extract files
        foreach ($part in $response.definition.parts) {
            $fileName = $part.path
            $filePath = Join-Path $tempDir $fileName
            $payloadType = $part.payloadType
            $payload = $part.payload
            
            if ($payloadType -eq "InlineBase64") {
                $bytes = [System.Convert]::FromBase64String($payload)
                $fileDir = Split-Path $filePath
                if (-not (Test-Path $fileDir)) {
                    New-Item -ItemType Directory -Path $fileDir -Force | Out-Null
                }
                [System.IO.File]::WriteAllBytes($filePath, $bytes)
            }
        }
        
        # Find the .pq file
        $pqFile = Get-ChildItem -Path $tempDir -Filter "*.pq" -Recurse | Select-Object -First 1
        
        if ($pqFile) {
            # Read the .pq file content
            $pqContent = Get-Content -Path $pqFile.FullName -Raw -Encoding UTF8
            
            # Save as .pqt
            $outputFileName = "$cleanWorkspaceName ~ $cleanDataflowName.pq"
            $outputFilePath = Join-Path $OutputPath $outputFileName
            $pqContent | Out-File -FilePath $outputFilePath -Encoding UTF8
                        
            # Clean up temp directory
            Remove-Item -Path $tempDir -Recurse -Force
            
            return @{
                Content = $pqContent
                FilePath = $outputFilePath
            }
        } else {
            Remove-Item -Path $tempDir -Recurse -Force
            return $null
        }
        
    } catch {
        return $null
    }
}

# Function to parse .pq content and extract queries
function Get-FabricDataflowQueries {
    param (
        [string]$Content,
        [string]$DataflowId,
        [string]$DataflowName,
        [string]$WorkspaceName,
        [datetime]$ReportDate
    )
    
    $queries = @()
    
    # Clean up names
    $cleanWorkspaceName = $WorkspaceName -replace '\[', '(' -replace '\]', ')'
    $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()
    
    $cleanDataflowName = $DataflowName -replace '\[', '(' -replace '\]', ')'
    $cleanDataflowName = $cleanDataflowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
    $cleanDataflowName = $cleanDataflowName.TrimStart()
    
    # Split content by "section Section1;" to get the queries section
    $sections = $Content -split 'section Section1;', 2
    
    if ($sections.Count -gt 1) {
        $queriesSection = $sections[1]
        
        # Use regex to find all queries:
        # Supports both shared QueryName = ... and shared #"Query Name With Spaces" = ...
        $pattern = '(?s)(?:\[[^\]]*\]\s*)?shared\s+(?:#"(.*?)"|([A-Za-z_]\w*))\s*=\s*(.*?)(?=(?:\[[^\]]*\]\s*)?shared\s+(?:#"(?:.*?)"|[A-Za-z_]\w*)\s*=|$)'
        $queryMatches = [regex]::Matches($queriesSection, $pattern)
        
        foreach ($match in $queryMatches) {
            # Group 1 = hash-quoted name, Group 2 = unquoted name, Group 3 = expression
            if ($match.Groups[1].Success) {
                $queryName = $match.Groups[1].Value
            } else {
                $queryName = $match.Groups[2].Value
            }
            $queryExpression = $match.Groups[3].Value.Trim()

            # Remove trailing semicolons
            $queryExpression = $queryExpression -replace ';\s*$', ''
            $queryExpression = $queryExpression.Trim()

            # Skip if empty or just whitespace
            if ([string]::IsNullOrWhiteSpace($queryName) -or [string]::IsNullOrWhiteSpace($queryExpression)) {
                continue
            }

            $queries += [PSCustomObject]@{
                "Dataflow ID" = $DataflowId
                "Dataflow Name" = $DataflowName
                "Query Name" = $queryName
                "Query" = $queryExpression
                "Report Date" = $ReportDate
                "Workspace Name - Dataflow Name" = "$cleanWorkspaceName ~ $cleanDataflowName"
            }
        }
    }
    
    return $queries
}

function Get-DataflowQueryConnectionDetails {
    param(
        [string]$QueryExpression
    )

    if ([string]::IsNullOrWhiteSpace($QueryExpression)) {
        return $null
    }

    function New-ConnectionSummaryJson {
        param(
            [hashtable]$Payload
        )

        return ($Payload | ConvertTo-Json -Compress)
    }

    # Manual-entered / embedded data patterns
    if ($QueryExpression -match '(?is)Binary\.FromText\(|Json\.Document\(\s*Binary\.|Table\.FromRows\(|#table\(') {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'manual-entered-data'
        })
    }

    # Dataflow-to-dataflow reference patterns (PowerPlatform.Dataflows navigation)
    $workspaceIdMatch = [regex]::Match($QueryExpression, '(?is)workspaceId\s*=\s*"([^"]+)"')
    $dataflowIdMatch = [regex]::Match($QueryExpression, '(?is)dataflowId\s*=\s*"([^"]+)"')
    if ($workspaceIdMatch.Success -or $dataflowIdMatch.Success -or $QueryExpression -match '(?is)PowerPlatform\.Dataflows|Dataflows\(') {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'dataflow'
            workspaceId = if ($workspaceIdMatch.Success) { $workspaceIdMatch.Groups[1].Value } else { $null }
            dataflowId = if ($dataflowIdMatch.Success) { $dataflowIdMatch.Groups[1].Value } else { $null }
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Sql\.Database\(\s*"([^"]+)"\s*,\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'sql'
            server = $m.Groups[1].Value.ToLowerInvariant()
            database = $m.Groups[2].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Oracle\.Database\(\s*"([^"]+)"')
    if ($m.Success) {
        $oracleNativeQueryMatch = [regex]::Match($QueryExpression, '(?is)\bQuery\s*=\s*"((?:[^"]|"")*)"')
        $oracleNativeQuery = if ($oracleNativeQueryMatch.Success) {
            $oracleNativeQueryMatch.Groups[1].Value -replace '""', '"'
        }
        else {
            $null
        }

        return (New-ConnectionSummaryJson -Payload @{
            type = 'oracle'
            server = $m.Groups[1].Value.ToLowerInvariant()
            nativeQuery = $oracleNativeQuery
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Excel\.Workbook\(\s*Web\.Contents\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'excel-web'
            url = $m.Groups[1].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Csv\.Document\(\s*Web\.Contents\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'csv-web'
            url = $m.Groups[1].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Web\.Contents\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'web'
            url = $m.Groups[1].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)OData\.Feed\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'odata'
            url = $m.Groups[1].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)SharePoint\.Files\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'sharepoint'
            site = $m.Groups[1].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Excel\.Workbook\(\s*File\.Contents\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'excel-file'
            file = $m.Groups[1].Value
        })
    }

    $m = [regex]::Match($QueryExpression, '(?is)Csv\.Document\(\s*File\.Contents\(\s*"([^"]+)"')
    if ($m.Success) {
        return (New-ConnectionSummaryJson -Payload @{
            type = 'csv-file'
            file = $m.Groups[1].Value
        })
    }

    return $null
}






if ($resumeMode -and $checkpoint.Phase6DataflowComplete) {
    Write-Output "[RESUME] Phase 6 (Dataflow) already complete. Skipping."
}

    $checkpoint.Phase6DataflowComplete = $true
    Save-Checkpoint
}
else {
Write-Output "Dataflow Backup and Detail Extraction Process Started"

# Define the dataflow backups path
$dataflowBackupsPath = Join-Path -Path $baseFolderPath -ChildPath "Dataflow Backups"

# Check if the "Dataflow Backups" folder exists, if not create it
if (-not (Test-Path -Path $dataflowBackupsPath)) {
    New-Item -Path $dataflowBackupsPath -ItemType Directory -Force | Out-Null
}

# Create a variable for end of week (Friday) date
$date = (Get-Date -UFormat "%Y-%m-%d")

# Create a new folder for the backups (clear only in overwrite mode)
$dataflow_new_date_folder = Join-Path -Path $dataflowBackupsPath -ChildPath $date
if ($script:OverwriteOutputs -and (Test-Path -Path $dataflow_new_date_folder)) {
    Write-Host "[INFO] Clearing existing dataflow backup folder for today: $dataflow_new_date_folder"
    Remove-Item -Path $dataflow_new_date_folder -Recurse -Force -ErrorAction SilentlyContinue
}
New-Item -Path $dataflow_new_date_folder -ItemType Directory -Force | Out-Null

# Set the base output file path
$baseOutputFilePath = $dataflow_new_date_folder

# Get the latest folder by date
$latestDataflowDateFolder = Get-ChildItem -Path $dataflowBackupsPath -Directory |
    Where-Object { $_.Name -match '^\d{4}-\d{2}-\d{2}$' } |  # Filter for folders with a date pattern
    Sort-Object { [datetime]::Parse($_.Name) } -Descending | # Sort by date, descending
    Select-Object -First 1

if ($latestDataflowDateFolder) {
    # Override $baseOutputFilePath with the latest-dated folder
    $folderPath = $latestDataflowDateFolder.FullName
    Write-Host "Using the latest dated folder: $baseOutputFilePath"
} else {
    # Throw an error if no valid dated folders are found
    Write-Error "No valid dated folders found in 'Dataflow Backups'."
    exit
}

# Set the combined Excel output path
$combinedExcelOutputPath = Join-Path -Path $dataflow_new_date_folder -ChildPath "Dataflow Detail.xlsx"

# Define the headers
$headers = @("Dataflow ID", "Dataflow Name", "Query Name", "Query", "Query Connection Details", "Report Date", "Workspace Name - Dataflow Name")

# Initialize a combined DataTable with the specified headers
$combinedDataTable = New-Object System.Data.DataTable
foreach ($header in $headers) {
   $null = $combinedDataTable.Columns.Add($header, [System.String])
}

# Get the current date
$currentDate = [datetime]::Parse($latestDataflowDateFolder.Name)


# Loop through all workspaces to fetch Gen1 dataflows (Power BI API)


foreach ($workspace in $workspacesInfo) {
    $workspaceName = $workspace.WorkspaceName
    $workspaceId = $workspace.WorkspaceId

    # Set the Power BI REST API URL for the dataflow details
    $dataflowDetailsUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$workspaceId/dataflows"

    # Get the list of dataflows in the workspace
    $dataflowsResponse = Invoke-PowerBIRestMethod -Url $dataflowDetailsUrl -Method Get

    # Parse the JSON response
    $dataflows = $dataflowsResponse | ConvertFrom-Json

    # Check if the response is valid and contains dataflows
    if ($dataflows -and $dataflows.value) {
        
        # Iterate through the Gen1 dataflows
        foreach ($dataflow in $dataflows.value) {
            $dataflowId = $dataflow.objectId
            $dataflowName = $dataflow.name

            # Clean up workspace name
            $cleanWorkspaceName = $workspaceName -replace '\[', '(' -replace '\]', ')'
            $cleanWorkspaceName = $cleanWorkspaceName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
            $cleanWorkspaceName = $cleanWorkspaceName.TrimStart()

            # Clean up dataflow name
            $cleanDataFlowName = $dataflowName -replace '\[', '(' -replace '\]', ')'
            $cleanDataFlowName = $cleanDataFlowName -replace "[^a-zA-Z0-9\(\)&,.-]", " "
            $cleanDataFlowName = $cleanDataFlowName.TrimStart()
            
            # Define output file path specific to the dataflow (Gen1)
            $dataflowOutputFilePath = Join-Path -Path $baseOutputFilePath -ChildPath "$cleanWorkspaceName ~ $cleanDataFlowName.txt"
            
            # Set the Power BI REST API URL for the specific dataflow
            $apiUrl = "$($global:PowerBIEndpoints.ApiPrefix)/v1.0/myorg/groups/$workspaceId/dataflows/$dataflowId"
            
            # Get the dataflow
            $response = Invoke-PowerBIRestMethod -Url $apiUrl -Method Get
            
            # Save the original response to text file for backup
            $response | Out-File -FilePath $dataflowOutputFilePath -Encoding UTF8
            
            # Parse the JSON response to extract document content
            try {
                $dataflowJson = $response | ConvertFrom-Json
                
                # Check if the JSON has the expected structure
                if (-not $dataflowJson.'pbi:mashup' -or -not $dataflowJson.'pbi:mashup'.document) {
                    continue
                }
                
                # Extract the document content (Power Query code)
                $documentContent = $dataflowJson.'pbi:mashup'.document
                
                # The document contains escaped content - unescape it
                # Replace escaped newlines and quotes
                $documentContent = $documentContent -replace '\\r\\n', "`n" -replace '\\n', "`n"
                $documentContent = $documentContent -replace '\\"', '"'
                
                # Split by "section Section1;" to get the queries section
                $sections = $documentContent -split 'section Section1;', 2
                
                if ($sections.Count -lt 2) {
                    continue
                }
                
                $queriesSection = $sections[1]
                
                # Use regex to extract queries (similar to Gen2 parsing for consistency)
                # Supports both shared QueryName = ... and shared #"Query Name With Spaces" = ...
                $queryPattern = '(?s)(?:\[[^\]]*\]\s*)?shared\s+(?:#"(.*?)"|([A-Za-z_]\w*))\s*=\s*(.*?)(?=(?:\[[^\]]*\]\s*)?shared\s+(?:#"(?:.*?)"|[A-Za-z_]\w*)\s*=|$)'
                $queryMatches = [regex]::Matches($queriesSection, $queryPattern)
                
                $data = @()
                foreach ($match in $queryMatches) {
                    # Group 1 = hash-quoted name, Group 2 = unquoted name, Group 3 = expression
                    if ($match.Groups[1].Success) {
                        $queryName = $match.Groups[1].Value
                    } else {
                        $queryName = $match.Groups[2].Value
                    }
                    $queryExpression = $match.Groups[3].Value.Trim()
                    
                    # Remove trailing semicolon if present
                    $queryExpression = $queryExpression -replace ';\s*$', ''
                    $queryExpression = $queryExpression.Trim()
                    
                    if ([string]::IsNullOrWhiteSpace($queryName) -or [string]::IsNullOrWhiteSpace($queryExpression)) {
                        continue
                    }
                    
                    $data += [PSCustomObject]@{
                        "Dataflow ID" = $dataflowId
                        "Dataflow Name" = $dataflowName
                        "Query Name" = $queryName
                        "Query" = $queryExpression
                        "Query Connection Details" = Get-DataflowQueryConnectionDetails -QueryExpression $queryExpression
                        "Report Date" = $currentDate
                        "Workspace Name - Dataflow Name" = "$cleanWorkspaceName ~ $cleanDataFlowName"
                    }
                }
                
                # Fill the combined DataTable with data
                foreach ($item in $data) {
                    $row = $combinedDataTable.NewRow()
                    $row["Dataflow ID"] = $item."Dataflow ID"
                    $row["Dataflow Name"] = $item."Dataflow Name"
                    $row["Query Name"] = $item."Query Name"
                    $row["Query"] = $item.Query
                    $row["Query Connection Details"] = $item."Query Connection Details"
                    $row["Report Date" ] = $item."Report Date"
                    $row["Workspace Name - Dataflow Name"] = $item."Workspace Name - Dataflow Name"
                    $combinedDataTable.Rows.Add($row)
                }
                
            } catch {
                continue
            }
        }
    } else {
    }
}


# Loop through all workspaces to fetch Gen2 dataflows (Fabric Items API)
foreach ($workspace in $workspacesInfo) {
    $workspaceName = $workspace.WorkspaceName
    $workspaceId = $workspace.WorkspaceId

    $fabricEndpoint = $global:PowerBIEndpoints.FabricApiPrefix
    $fabricItemsSupported = ($fabricEndpoint -match '(?i)^https://api\.fabric\.microsoft\.com$')

    if (-not $fabricItemsSupported) {
        continue
    }

    # Set the Fabric API URL for items
    $fabricItemsUrl = "$fabricEndpoint/v1/workspaces/$workspaceId/items"

    try {
        $fabricHeaders = @{
            'Authorization' = "Bearer $(Get-CurrentAccessToken)"
            'Content-Type'  = 'application/json'
        }

        # Get the list of items in the workspace
        $fabricItemsResponse = Invoke-AuthenticatedRestMethodWithRetry -Uri $fabricItemsUrl -Headers $fabricHeaders -Method GET

        # Filter for Dataflow items (Gen2 dataflows)
        $gen2Dataflows = $fabricItemsResponse.value | Where-Object { $_.type -eq 'Dataflow' }

        if ($gen2Dataflows -and $gen2Dataflows.Count -gt 0) {

            # Iterate through the Gen2 dataflows
            foreach ($dataflow in $gen2Dataflows) {
                $dataflowId = $dataflow.id
                $dataflowName = $dataflow.displayName


                # Get current access token
                $currentAccessToken = Get-CurrentAccessToken

                # Download and extract Fabric dataflow
                $fabricResult = Export-FabricDataflow -WorkspaceId $workspaceId `
                                                      -DataflowId $dataflowId `
                                                      -WorkspaceName $workspaceName `
                                                      -DataflowName $dataflowName `
                                                      -OutputPath $baseOutputFilePath `
                                                      -AccessToken $currentAccessToken

                if ($fabricResult) {
                    # Parse the .pq content
                    $parsedQueries = Get-FabricDataflowQueries -Content $fabricResult.Content `
                                                                  -DataflowId $dataflowId `
                                                                  -DataflowName $dataflowName `
                                                                  -WorkspaceName $workspaceName `
                                                                  -ReportDate $currentDate

                    # Add parsed queries to combined DataTable
                    foreach ($query in $parsedQueries) {
                        $row = $combinedDataTable.NewRow()
                        $row["Dataflow ID"] = Convert-ToExcelSafeText -Value $query."Dataflow ID"
                        $row["Dataflow Name"] = Convert-ToExcelSafeText -Value $query."Dataflow Name"
                        $row["Query Name"] = Convert-ToExcelSafeText -Value $query."Query Name"
                        $row["Query"] = Convert-ToExcelSafeText -Value $query.Query
                        $row["Query Connection Details"] = Convert-ToExcelSafeText -Value (Get-DataflowQueryConnectionDetails -QueryExpression $query.Query)
                        $row["Report Date"] = Convert-ToExcelSafeText -Value $query."Report Date"
                        $row["Workspace Name - Dataflow Name"] = Convert-ToExcelSafeText -Value $query."Workspace Name - Dataflow Name"
                        $combinedDataTable.Rows.Add($row)
                    }
                }
            }
        }
    }
    catch {
        continue
    }
}

# Check if the combined DataTable has any rows, if not add a dummy row with headers only
if ($combinedDataTable.Rows.Count -eq 0) {

$combinedDataTable | Export-Excel -Path $combinedExcelOutputPath -AutoSize
Write-Host "Data exported to $combinedExcelOutputPath"

if (Test-Path -Path $sourceFilePath) {

Save-Checkpoint

}
